57def _load_merged_config(config_path: str, preset_override: Optional[str] = None) -> dict:
58 """@brief Resolve the layered configuration: master + emergent preset + per-run override.
59
60 Three layers are merged, each overriding the previous:
61 1. the master config.json (globals plus the @c emergent_preset selector);
62 2. the emergent preset it names (`configs/emergent/<name>.json`), if any;
63 3. the per-run --config file (used by run_all.py), if different from master.
64 The @c pair_rules table is deep-merged at each step; comment keys (those
65 starting with '_') are dropped.
66
67 @param config_path Path to the per-run config (may equal the master path).
68 @param preset_override If given, overrides the master's @c emergent_preset selector
69 (used by the --preset CLI flag to batch over presets).
70 @return dict the fully merged configuration.
71 """
72 def _read(path: str) -> dict:
73 try:
74 with open(path, "r") as f:
75 raw = json.load(f)
76 return {k: v for k, v in raw.items() if not k.startswith("_")}
77 except (FileNotFoundError, json.JSONDecodeError):
78 return {}

80 def _deep_merge(base: dict, override: dict) -> dict:
81 """Shallow-merge two configs, but deep-merge their pair_rules tables."""
82 merged = {**base, **override}
83 base_rules = base.get("pair_rules", {})
84 ovr_rules = override.get("pair_rules", {})
85 if base_rules or ovr_rules:
86 merged["pair_rules"] = {**base_rules, **ovr_rules}
87 return merged
88
89 cfg = _read(DEFAULT_CONFIG_PATH)
90
91 # Overlay the named emergent preset so switching `emergent_preset` in
92 # config.json swaps the whole attract/repel personality in one place.
93 preset_name = preset_override if preset_override is not None else cfg.get("emergent_preset")
94 if preset_name:
95 preset_path = os.path.join(
96 os.path.dirname(DEFAULT_CONFIG_PATH), "configs", "emergent", f"{preset_name}.json"
97 )
98 preset = _read(preset_path)
99 if preset:
100 cfg = _deep_merge(cfg, preset)
101 else:
102 print(f"[config] emergent_preset {preset_name!r} not found at {preset_path} — ignoring.")
103
104 # Overlay the per-run --config file (run_all.py ablation), if any.
105 if os.path.abspath(config_path) == os.path.abspath(DEFAULT_CONFIG_PATH):
106 return cfg
107 override = _read(config_path)
108 if not override:
109 return cfg
110 return _deep_merge(cfg, override)

113def _load_force_config(config_path: str, preset_override: Optional[str] = None) -> Tuple[float, float, Dict, float]:
114 """@brief Load g_attract, g_repel and the optional pair_rules from merged config.
115
116 Config force values are in a readable ~-1..+1 scale; @c force_scale multiplies
117 them into internal units before use. Falls back to hardcoded defaults if the
118 config is missing or malformed.
119
120 @param config_path Path to the per-experiment config.
121 @param preset_override If given, overrides the master's @c emergent_preset selector
122 (passed through to _load_merged_config; used by the --preset CLI flag).
123 @return tuple (g_attract, g_repel, rules, force_scale) where @c rules maps (i, j) -> float.
124 """
125 FALLBACK_SCALE = 25.0
126 FALLBACK_ATTRACT = -10.0 / FALLBACK_SCALE # = -0.4
127 FALLBACK_REPEL = 25.0 / FALLBACK_SCALE # = 1.0
128
129 cfg = _load_merged_config(config_path, preset_override)
130 if not cfg:
131 print("[config] config.json not found — using hardcoded defaults.")
132 return -10.0, 25.0, {}, FALLBACK_SCALE
133
134 scale = float(cfg.get("force_scale", FALLBACK_SCALE))
135 g_attract = float(cfg.get("g_attract", FALLBACK_ATTRACT)) * scale
136 g_repel = float(cfg.get("g_repel", FALLBACK_REPEL)) * scale
137
138 rules: Dict[Tuple, float] = {}
139 for key, val in cfg.get("pair_rules", {}).items():
140 try:
141 i, j = (int(x) for x in key.split("-"))
142 internal_val = float(val) * scale
143 rules[(min(i, j), max(i, j))] = internal_val #max(-50.0, min(50.0, internal_val))
144 except (ValueError, TypeError):
145 print(f"[config] Skipping invalid pair_rules entry: {key!r}: {val!r}")
146
147 print(f"[config] Loaded from config.json — force_scale={scale}, "
148 f"g_attract={g_attract:.2f}, g_repel={g_repel:.2f}, "
149 f"{len(rules)} pair rule(s)")
150 return g_attract, g_repel, rules, scale

153def apply_physics_rules(
154 particles: List[Particle],
155 obstacles: List[Tuple[int, int, int]],
156 g_attract: float,
157 g_repel: float,
158 dt: float,
159 rules: dict = None,
160 attraction_enabled: bool = True):
161 """@brief Integrate one physics tick over the agents (the physics 'system').
162
163 The simulation-side glue around the pure @ref physics force library: it
164 loops over Particle objects, resolves each pair's force via
165 @ref physics.pair_force (emergency repulsion + the standard gravity formula),
166 adds soft-obstacle and model-directed behavioral pushes, then integrates
167 velocities/positions and resolves hard obstacle and wall collisions.
168
169 @param particles List of all Particle agents; positions/velocities mutated in place.
170 @param obstacles List of (ox, oy, radius) obstacle circles.
171 @param g_attract Default intra-cluster coefficient (negative attracts).
172 @param g_repel Default inter-cluster coefficient (positive repels).
173 @param dt Integration timestep in seconds.
174 @param rules Optional dict (min(ci,cj), max(ci,cj)) -> coefficient for per-pair forces.
175 @param attraction_enabled If False, intra/inter-cluster forces are disabled and only the
176 emergency anti-stacking repulsion remains — used to ablate the emergent physics.
177 @return None.
178 """
179 n = len(particles)
180 forces = [np.zeros(2) for _ in particles]
181
182 # --- Pairwise inter-agent forces (emergency repulsion + standard gravity) ---
183 upper_sq = PARTICLE_FORCE_UPPER_RANGE ** 2
184 for i in range(n):
185 a = particles[i]
186 for j in range(i + 1, n):
187 b = particles[j]
188
189 dx = a.x - b.x
190 dy = a.y - b.y
191 d_sq = dx ** 2 + dy ** 2
192
193 # Cull far pairs before any sqrt / force math.
194 if d_sq > upper_sq:
195 continue
196 if d_sq == 0:
197 d_sq = 0.001
198 d = d_sq ** 0.5
199
200 coeff = physics.interaction_coefficient(
201 a.cluster_id, b.cluster_id, g_attract, g_repel, rules)
202 f_scalar = physics.pair_force(
203 d, coeff, n, PARTICLE_FORCE_LOWER_RANGE,
204 power=PARTICLE_POWER_OF_DISTANCE,
205 attraction_enabled=attraction_enabled,
206 )
207
208 fx = f_scalar * dx
209 fy = f_scalar * dy
210 forces[i] += np.array([fx, fy])
211 forces[j] -= np.array([fx, fy])
212
213 # --- Soft obstacle zone: pre-contact repulsion so agents redirect early ---
214 soft_zone = cfl_params.get_obstacle_soft_zone()
215 soft_strength = cfl_params.get_obstacle_soft_strength()
216 for i, p in enumerate(particles):
217 for ox, oy, orad in obstacles:
218 dx, dy = p.x - ox, p.y - oy
219 dist = math.hypot(dx, dy)
220 if dist == 0:
221 continue
222 push = physics.soft_obstacle_push(dist, orad + p.r, soft_zone, soft_strength)
223 forces[i][0] += (dx / dist) * push
224 forces[i][1] += (dy / dist) * push
225
226 # --- Behavioral force: model-directed push scaled by confidence ---
227 # model[3] (obstacle_pressure) amplifies the force so sustained pressure
228 # translates directly into a stronger escape push.
229 behavioral_force = cfl_params.get_behavioral_force()
230 for i, p in enumerate(particles):
231 if p.cluster_id != -1 and p.model[2] > 0.15:
232 pressure_boost = 1.0 + float(p.model[3]) * 1.25
233 strength = behavioral_force * float(p.model[2]) * pressure_boost
234 forces[i][0] += p.model[0] * strength
235 forces[i][1] += p.model[1] * strength
236
237 # --- Integrate velocities/positions, resolve obstacle + wall collisions ---
238 for i, p in enumerate(particles):
239 fx, fy = forces[i]
240
241 p.vx = (p.vx + fx * dt) * PARTICLE_LOSE_ENERGY
242 p.vy = (p.vy + fy * dt) * PARTICLE_LOSE_ENERGY
243 p.vx, p.vy = physics.clamp_speed(p.vx, p.vy, PARTICLE_MAX_SPEED)
244
245 p.x += p.vx * dt
246 p.y += p.vy * dt
247
248 # Hard physical collisions with obstacles: push out + bounce.
249 for ox, oy, orad in obstacles:
250 dx, dy = p.x - ox, p.y - oy
251 dist = math.hypot(dx, dy)
252 if dist < orad + p.r and dist > 0:
253 overlap = (orad + p.r) - dist
254 nx, ny = dx / dist, dy / dist
255 p.x += nx * overlap
256 p.y += ny * overlap
257 p.vx, p.vy = physics.reflect_velocity(p.vx, p.vy, nx, ny, restitution=0.8)
258
259 # Wall collisions (simulation bounds, not the window).
260 V = 0.9
261 D = WALL_BOUNDARY
262 if p.x < D:
263 p.x = D
264 p.vx *= -V
265 if p.x > SIM_DIM[0] - D:
266 p.x = SIM_DIM[0] - D
267 p.vx *= -V
268 if p.y < D:
269 p.y = D
270 p.vy *= -V
271 if p.y > SIM_DIM[1] - D:
272 p.y = SIM_DIM[1] - D
273 p.vy *= -V

340 @staticmethod
341 def _init_obstacles(spec):
342 """@brief Resolve the @c obstacles config field into a list of (x, y, r) tuples.
343
344 Accepted forms: None/missing -> 4 random; int N -> N random; list of
345 [x, y, r] or {"x":..,"y":..,"r":..} -> exactly those. Malformed entries are
346 skipped with a warning.
347
348 @param spec The raw @c obstacles value from the config.
349 @return list of (x, y, radius) obstacle tuples.
350 """
351 def _random_set(n):
352 return [
353 (random.randint(150, SIM_WIDTH - 150),
354 random.randint(150, SIM_HEIGHT - 150),
355 random.randint(30, 70))
356 for _ in range(n)
357 ]
358
359 if spec is None:
360 print("[config] obstacles: random (4)")
361 return _random_set(4)
362
363 if isinstance(spec, int):
364 n = max(0, spec)
365 print(f"[config] obstacles: random ({n})")
366 return _random_set(n)
367
368 if not isinstance(spec, list):
369 print(f"[config] obstacles: invalid type {type(spec).__name__} — using random (4)")
370 return _random_set(4)
371
372 out = []
373 for idx, item in enumerate(spec):
374 try:
375 if isinstance(item, dict):
376 x, y, r = int(item["x"]), int(item["y"]), int(item["r"])
377 elif isinstance(item, (list, tuple)) and len(item) == 3:
378 x, y, r = int(item[0]), int(item[1]), int(item[2])
379 else:
380 raise ValueError("expected [x,y,r] or {x,y,r}")
381 if r <= 0:
382 raise ValueError("r must be positive")
383 out.append((x, y, r))
384 except (KeyError, ValueError, TypeError) as e:
385 print(f"[config] obstacles[{idx}] skipped: {item!r} ({e})")
386
387 print(f"[config] obstacles: explicit ({len(out)} from config.json)")
388 return out

390 def _init_sim(self):
391 """@brief Build the initial world from config: clusters, agents, obstacles, forces, logger.
392
393 Spawns one populated group per cluster when @c num_clusters is configured
394 (skipping the initial CFL round), or five randomly-assigned groups settled by
395 one CFL round otherwise. Also seeds targets, trails, colors and the SimLogger.
396 """
397 _cfg = _load_merged_config(self.config_path, self.preset)
398 _nc = _cfg.get("num_clusters")
399 _min_clusters = cfl_params.get_min_clusters()
400 _max_clusters = cfl_params.get_max_clusters()
401
402 if isinstance(_nc, int) and _min_clusters <= _nc <= _max_clusters:
403 self.num_clusters = _nc
404 print(f"[config] num_clusters={self.num_clusters} (from config)")
405 else:
406 self.num_clusters = random.randint(_min_clusters, _max_clusters)
407 if _nc is not None:
408 print(f"[config] num_clusters={_nc!r} out of range [{_min_clusters},{_max_clusters}] — using random ({self.num_clusters})")
409
410 D = WALL_BOUNDARY * 4
411
412 # Per-true-cluster spatial anchors: evenly spaced on a circle inside the
413 # sim. All members of true cluster i share anchor i as their _target_x/_y
414 # so cluster members have a SHARED goal — that shared signal is what CFL
415 # aggregation can recover, while individual learning cannot.
416 _cx, _cy = SIM_WIDTH / 2.0, SIM_HEIGHT / 2.0
417 _radius = min(SIM_WIDTH, SIM_HEIGHT) / 2.0 - D
418 self._true_anchors: List[Tuple[float, float]] = [
419 (_cx + _radius * math.cos(2 * math.pi * i / self.num_clusters),
420 _cy + _radius * math.sin(2 * math.pi * i / self.num_clusters))
421 for i in range(self.num_clusters)
422 ]
423 self._true_anchor_angles = [random.uniform(0, 2 * math.pi) for _ in range(self.num_clusters)]
424
425 # Pin initial cluster_targets to the true anchors so IFCA has a
426 # non-degenerate spatial loss landscape from round 1 (otherwise the
427 # random init scrambles them to ~the sim centroid).
428 self.cluster_targets = [(int(ax), int(ay)) for ax, ay in self._true_anchors]
429
430 # Shared anchors pin K — disable IFCA split/merge from the start.
431 self.cooldown_counter = 10**6
432 self.cluster_ages = {i: cfl_params.get_blend_maturity_rounds() for i in range(self.num_clusters)}
433
434 trail_maxlen = FRAME_RATE * 3 # ~3 seconds of history at FRAME_RATE sample rate
435 self.target_trails: List[deque] = [deque(maxlen=trail_maxlen) for _ in range(self.num_clusters)]
436 self._trail_counter = 0
437
438 self.cluster_colors = {i: CLUSTER_PALETTE[i] for i in range(self.num_clusters)}
439 self.cluster_colors[-1] = (80, 80, 80)
440
441 # Spawn one group per true cluster. Each group shares a latent bias
442 # direction (evenly spaced around 2π) AND a shared spatial anchor —
443 # together those constitute the ground-truth latent structure CFL is
444 # meant to recover. Groups spawn in the global center box, but each has
445 # _target_x/_y pinned to its own perimeter anchor — so particles start
446 # far from their target, geometric loss is large at round 1, and the
447 # loss curve shows a real per-round decrease as the swarm navigates to
448 # its anchor (spawning AT the anchor would flatten the curve at the
449 # bias-limited floor). Initial cluster_id is randomized so init purity
450 # ≈ 1/K — CFL must do the discovery work, CFL-off stays at the baseline.
451 _cluster_bias_dirs = [
452 (math.cos(2 * math.pi * i / self.num_clusters),
453 math.sin(2 * math.pi * i / self.num_clusters))
454 for i in range(self.num_clusters)
455 ]
456
457 self.all_particles = []
458 for i in range(self.num_clusters):
459 ax, ay = self._true_anchors[i]
460 group = instantiate_group(
461 num=PARTICLE_DEFAULT_SPAWN_NUM,
462 c=PARTICLE_COLOR_WHITE,
463 frame=PARTICLE_DEFAULT_SPAWN_FRAME,
464 target_idx=i,
465 true_cluster_id=i,
466 cluster_bias_dir=_cluster_bias_dirs[i],
467 )
468 for p in group:
469 p._target_x = float(ax)
470 p._target_y = float(ay)
471 p._target_angle = self._true_anchor_angles[i]
472 p.cluster_id = random.randint(0, self.num_clusters - 1)
473 self.all_particles.extend(group)
474
475 self.kmeans = KMeans(n_clusters=self.num_clusters, n_init=10, random_state=0)
476
477 # Physics steps between CFL rounds. Shorter → model isn't fully converged
478 # between rounds, so the per-round loss trajectory shows real convergence
479 # across rounds (not just a flat line at the bias-limited floor).
480 # Configurable via config.json's "cluster_update_interval".
481 _cui = _cfg.get("cluster_update_interval")
482 self.cluster_update_interval = int(_cui) if isinstance(_cui, (int, float)) and _cui > 0 else 50
483
484 # Rounds during which IFCA split/merge is suppressed (only relevant in
485 # shared-anchor mode where K is pinned to K_true). Lets IFCA find the
486 # latent structure first, then unpins so dynamic-K behavior can emerge.
487 # Configurable via config.json's "split_merge_start_round". 0 = never
488 # pin; very large = always pin.
489 _smsr = _cfg.get("split_merge_start_round")
490 self.split_merge_start_round = int(_smsr) if isinstance(_smsr, (int, float)) and _smsr >= 0 else 20
491 self.cluster_update_timer = 0
492 self.cfl_round_counter = 0
493 self.cfl_enabled = bool(_cfg.get("cfl_enabled", True))
494 self.attraction_enabled = bool(_cfg.get("attraction_enabled", True))
495 _mr = _cfg.get("max_rounds")
496 self.max_rounds = int(_mr) if isinstance(_mr, (int, float)) and _mr > 0 else None
497 self._avg_loss_history: deque = deque(maxlen=60)
498 self._avg_conf_history: deque = deque(maxlen=60)
499
500 self.obstacles = self._init_obstacles(_cfg.get("obstacles"))
501
502 self.g_attract, self.g_repel, self.rules, self.force_scale = _load_force_config(self.config_path, self.preset)
503
504 self._sync_rules()
505
506 print(f"\n{'=' * 60}")
507 print(f"INIT: Starting Simulation with {len(self.all_particles)} particles "
508 f"in {self.num_clusters} clusters.")
509 print(f"{'=' * 60}\n")
510
511 _tags = []
512 if "cfl_enabled" in _cfg and _cfg["cfl_enabled"]:
513 _tags.append("cfl")
514 if "attraction_enabled" in _cfg and _cfg["attraction_enabled"]:
515 _tags.append("emergent")
516 self.logger = SimLogger(
517 log_dir=self.log_dir,
518 tags="_".join(_tags),
519 cfl_enabled=self.cfl_enabled,
520 attraction_enabled=self.attraction_enabled,
521 run_name=self.run_name,
522 )

528 def _sync_rules(self):
529 """@brief Add missing default rules for current cluster pairs and drop stale ones."""
530 current_keys = set()
531 for i in range(self.num_clusters):
532 for j in range(i, self.num_clusters):
533 key = (i, j)
534 current_keys.add(key)
535 if key not in self.rules:
536 self.rules[key] = self.g_attract if i == j else self.g_repel
537 for k in [k for k in self.rules if k not in current_keys]:
538 del self.rules[k]

547 @param i First cluster id. @param j Second cluster id. @param val Internal-scale force.

597 def _step(self, dt: float):
598 """@brief Advance the simulation by one physics tick.
599
600 Drifts each agent's inner target, runs local_train() for every agent,
601 fires an IFCA run_cfl_round() when the round timer elapses (handling
602 split/merge bookkeeping, metrics and logging), then applies
603 update_peer_alignment() and apply_physics_rules() and samples trails.
604
605 @param dt Integration timestep in seconds.
606 """
607 _DRIFT_SPEED = 0.10
608 _DRIFT_MARGIN = WALL_BOUNDARY * 4
609
610 # Drift K anchors (one per true cluster), then refresh each particle's
611 # _target from its true cluster's anchor. This keeps all members of a
612 # true cluster locked to a SHARED goal — federation can cancel the
613 # remaining per-particle bias noise; individual learning cannot.
614 for i in range(len(self._true_anchors)):
615 ang = self._true_anchor_angles[i] + random.uniform(-0.01, 0.01)
616 ax, ay = self._true_anchors[i]
617 nx = ax + _DRIFT_SPEED * math.cos(ang)
618 ny = ay + _DRIFT_SPEED * math.sin(ang)
619 if nx < _DRIFT_MARGIN or nx > SIM_WIDTH - _DRIFT_MARGIN:
620 ang = math.pi - ang
621 nx = max(_DRIFT_MARGIN, min(SIM_WIDTH - _DRIFT_MARGIN, nx))
622 if ny < _DRIFT_MARGIN or ny > SIM_HEIGHT - _DRIFT_MARGIN:
623 ang = -ang
624 ny = max(_DRIFT_MARGIN, min(SIM_HEIGHT - _DRIFT_MARGIN, ny))
625 self._true_anchors[i] = (nx, ny)
626 self._true_anchor_angles[i] = ang
627
628 for p in self.all_particles:
629 if 0 <= p._true_cluster_id < len(self._true_anchors):
630 p._target_x, p._target_y = self._true_anchors[p._true_cluster_id]
631
632 # Pin cluster_targets[k] = anchor_k for k < K_true so the IFCA loss
633 # landscape stays aligned with the ground truth. Any extra entries
634 # introduced by IFCA splits keep whatever value run_cfl_round set.
635 for i in range(min(len(self.cluster_targets), len(self._true_anchors))):
636 self.cluster_targets[i] = self._true_anchors[i]
637
638 # Compute per-particle emergent pull (net attract/repel from neighbors)
639 # so local_train can fold it into the learned heading. Same force topology
640 # as apply_physics_rules — respects the rules table, falls back to
641 # g_attract/g_repel by intra/inter-cluster. Skipped when attraction is
642 # off so the emergent_pull defaults to (0,0) and behavior is identical
643 # to the pre-emergent-model code path.
644 n_parts = len(self.all_particles)
645 emergent_pulls = [(0.0, 0.0)] * n_parts
646 if self.attraction_enabled and n_parts > 1:
647 from src.constants import PARTICLE_FORCE_UPPER_RANGE
648 _upper_sq = PARTICLE_FORCE_UPPER_RANGE ** 2
649 for i, p in enumerate(self.all_particles):
650 if p.cluster_id == -1:
651 continue
652 ex, ey = 0.0, 0.0
653 for j, other in enumerate(self.all_particles):
654 if i == j or other.cluster_id == -1:
655 continue
656 dx = p.x - other.x
657 dy = p.y - other.y
658 d_sq = dx * dx + dy * dy
659 if d_sq > _upper_sq or d_sq < 1.0:
660 continue
661 ci, cj = p.cluster_id, other.cluster_id
662 if self.rules:
663 g = self.rules.get((min(ci, cj), max(ci, cj)),
664 self.g_attract if ci == cj else self.g_repel)
665 else:
666 g = self.g_attract if ci == cj else self.g_repel
667 d = math.sqrt(d_sq)
668 F = g / (d * n_parts)
669 ex += F * dx
670 ey += F * dy
671 emergent_pulls[i] = (ex, ey)
672
673 # Local training — each particle chases its own inner target
674 for i, p in enumerate(self.all_particles):
675 local_train(p, (p._target_x, p._target_y), self.obstacles,
676 learning_rate=0.02, emergent_pull=emergent_pulls[i])
677
678 # Simulation round — fires every cluster_update_interval physics steps.
679 self.cluster_update_timer += 1
680 if self.cluster_update_timer >= self.cluster_update_interval:
681 self.cluster_update_timer = 0
682 self.cfl_round_counter += 1
683
684 if self.max_rounds is not None and self.cfl_round_counter >= self.max_rounds:
685 print(f"\n[SIM] max_rounds={self.max_rounds} reached — stopping simulation.")
686 self._stop_event.set()
687 pygame.event.post(pygame.event.Event(pygame.QUIT))
688 return
689
690 # Track metrics every round regardless of whether CFL is enabled,
691 # so the sparkline shows the contrast when the mode is toggled.
692 if self.all_particles:
693 _al = sum(p.model[6] for p in self.all_particles) / len(self.all_particles)
694 _ac = sum(p.model[2] for p in self.all_particles) / len(self.all_particles)
695 self._avg_loss_history.append(_al)
696 self._avg_conf_history.append(_ac)
697
698 if self.cfl_enabled:
699 transfers, self.kmeans, self.cluster_targets, self.cluster_colors, \
700 self.cluster_ages, self.num_clusters, event, self.cooldown_counter = run_cfl_round(
701 self.all_particles, self.kmeans, self.cluster_targets,
702 self.cluster_colors, self.cluster_ages, self.cooldown_counter,
703 )
704
705 # Shared-anchor mode pins K=K_true for the first
706 # `split_merge_start_round` rounds so IFCA can find the latent
707 # structure cleanly. After that, the pin is released and IFCA's
708 # split/merge becomes active so K can drift adaptively. Without
709 # the explicit reset, the leftover 999_999 from the pin would
710 # take a million rounds to decrement back to 0.
711 if self._true_anchors:
712 if self.cfl_round_counter < self.split_merge_start_round:
713 self.cooldown_counter = 10**6
714 elif self.cooldown_counter > 100:
715 self.cooldown_counter = 0
716
717 max_idx = len(self.cluster_targets) - 1
718 for p in self.all_particles:
719 if p.cluster_id >= 0:
720 p.target_idx = min(p.cluster_id, max_idx)
721
722 event_type = event[0] if isinstance(event, tuple) else event
723 trail_maxlen = self.target_trails[0].maxlen if self.target_trails else FRAME_RATE * 3
724
725 if event_type == 'split':
726 self.target_trails.append(deque(maxlen=trail_maxlen))
727 self._sync_rules()
728 print(f" > SPLIT — now {self.num_clusters} clusters")
729 elif event_type == 'merge':
730 drop_idx = event[1]
731 if drop_idx < len(self.target_trails):
732 del self.target_trails[drop_idx]
733 elif self.target_trails:
734 self.target_trails.pop()
735 self._sync_rules()
736 print(f" > MERGE — now {self.num_clusters} clusters")
737
738 inertia = self.kmeans.inertia_ if hasattr(self.kmeans, 'inertia_') else 0.0
739 counts = {}
740 for p in self.all_particles:
741 counts[p.cluster_id] = counts.get(p.cluster_id, 0) + 1
742
743 print(f"\n[ROUND {self.cfl_round_counter}] CFL Complete")
744 print(f" > Inertia: {inertia:.2f}")
745 print(f" > Cluster Sizes: {dict(sorted(counts.items()))}")
746 if transfers:
747 print(f" > Migrations:")
748 for (old_id, new_id), cnt in sorted(transfers.items(), key=lambda x: -x[1]):
749 src = "Unassigned" if old_id == -1 else f"Cluster {old_id}"
750 print(f" - {cnt:3d} agents: {src} -> Cluster {new_id}")
751 else:
752 print(" > Migrations: (Stable)")
753 print("-" * 50)
754
755 self.logger.log_round(
756 round_num=self.cfl_round_counter,
757 particles=self.all_particles,
758 kmeans=self.kmeans,
759 cluster_targets=self.cluster_targets,
760 transfers=transfers,
761 event=event_type,
762 num_clusters=self.num_clusters,
763 )
764 else:
765 print(f"\n[ROUND {self.cfl_round_counter}] (CFL disabled — individual learning only)")
766 self.logger.log_round(
767 round_num=self.cfl_round_counter,
768 particles=self.all_particles,
769 kmeans=self.kmeans,
770 cluster_targets=self.cluster_targets,
771 transfers={},
772 event=None,
773 num_clusters=self.num_clusters,
774 )
775
776 if self.cfl_round_counter % 20 == 0:
777 self.logger.plot_all()
778
779 update_peer_alignment(self.all_particles)
780 apply_physics_rules(
781 self.all_particles, self.obstacles, self.g_attract, self.g_repel,
782 dt, self.rules, attraction_enabled=self.attraction_enabled,
783 )
784
785 # Trail sampling at reduced rate
786 self._trail_counter += 1
787 if self._trail_counter >= TRAIL_SAMPLE_EVERY:
788 self._trail_counter = 0
789 while len(self.target_trails) < len(self.cluster_targets):
790 maxlen = self.target_trails[0].maxlen if self.target_trails else FRAME_RATE * 3
791 self.target_trails.append(deque(maxlen=maxlen))
792 while len(self.target_trails) > len(self.cluster_targets):
793 self.target_trails.pop()
794 for i, (tx, ty) in enumerate(self.cluster_targets):
795 self.target_trails[i].append((int(tx), int(ty)))

852 def run(self):
853 """@brief Thread entry point: drain commands, step, publish, and pace to PHYSICS_HZ until stopped."""
854 dt = 1.0 / FRAME_RATE # physics uses same dt the old code used per-step
855 period = 1.0 / PHYSICS_HZ
856
857 try:
858 while not self._stop_event.is_set():
859 t0 = time.perf_counter()
860
861 # Drain commands (all of them before the next physics step)
862 while True:
863 try:
864 self._handle_cmd(self.cmd_queue.get_nowait())
865 except queue.Empty:
866 break
867
868 self._step(dt)
869
870 snap = self._build_snapshot()
871 with self._snapshot_lock:
872 self._snapshot = snap
873
874 elapsed = time.perf_counter() - t0
875 remaining = period - elapsed
876 if remaining > 0.0005:
877 time.sleep(remaining)
878 finally:
879 self.logger.close()
