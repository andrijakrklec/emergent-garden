51def _load_merged_config(config_path: str, preset_override: Optional[str] = None) -> dict:
52 """@brief Resolve the layered configuration: master + emergent preset + per-run override.
53
54 Three layers are merged, each overriding the previous:
55 1. the master config.json (globals plus the @c emergent_preset selector);
56 2. the emergent preset it names (configs/emergent/<name>.json), if any;
57 3. the per-run --config file (used by run_all.py), if different from master.
58 The @c pair_rules table is deep-merged at each step; comment keys (those
59 starting with '_') are dropped.
60
61 @param config_path Path to the per-run config (may equal the master path).
62 @param preset_override If given, overrides the master's @c emergent_preset selector
63 (used by the --preset CLI flag to batch over presets).
64 @return dict the fully merged configuration.
65 """
66 def _read(path: str) -> dict:
67 try:
68 with open(path, "r") as f:
69 raw = json.load(f)
70 return {k: v for k, v in raw.items() if not k.startswith("_")}
71 except (FileNotFoundError, json.JSONDecodeError):
72 return {}

74 def _deep_merge(base: dict, override: dict) -> dict:
75 """Shallow-merge two configs, but deep-merge their pair_rules tables."""
76 merged = {**base, **override}
77 base_rules = base.get("pair_rules", {})
78 ovr_rules = override.get("pair_rules", {})
79 if base_rules or ovr_rules:
80 merged["pair_rules"] = {**base_rules, **ovr_rules}
81 return merged
82
83 cfg = _read(DEFAULT_CONFIG_PATH)
84
85 # Overlay the named emergent preset so switching `emergent_preset` in
86 # config.json swaps the whole attract/repel personality in one place.
87 preset_name = preset_override if preset_override is not None else cfg.get("emergent_preset")
88 if preset_name:
89 preset_path = os.path.join(
90 os.path.dirname(DEFAULT_CONFIG_PATH), "configs", "emergent", f"{preset_name}.json"
91 )
92 preset = _read(preset_path)
93 if preset:
94 cfg = _deep_merge(cfg, preset)
95 else:
96 print(f"[config] emergent_preset {preset_name!r} not found at {preset_path} — ignoring.")
97
98 # Overlay the per-run --config file (run_all.py ablation), if any.
99 if os.path.abspath(config_path) == os.path.abspath(DEFAULT_CONFIG_PATH):
100 return cfg
101 override = _read(config_path)
102 if not override:
103 return cfg
104 return _deep_merge(cfg, override)

107def _load_force_config(config_path: str, preset_override: Optional[str] = None) -> Tuple[float, float, Dict, float]:
108 """@brief Load g_attract, g_repel and the optional pair_rules from merged config.
109
110 Config force values are in a readable ~-1..+1 scale; @c force_scale multiplies
111 them into internal units before use. Falls back to hardcoded defaults if the
112 config is missing or malformed.
113
114 @param config_path Path to the per-experiment config.
115 @return tuple (g_attract, g_repel, rules, force_scale) where @c rules maps (i, j) -> float.
116 """
117 FALLBACK_SCALE = 25.0
118 FALLBACK_ATTRACT = -10.0 / FALLBACK_SCALE # = -0.4
119 FALLBACK_REPEL = 25.0 / FALLBACK_SCALE # = 1.0
120
121 cfg = _load_merged_config(config_path, preset_override)
122 if not cfg:
123 print("[config] config.json not found — using hardcoded defaults.")
124 return -10.0, 25.0, {}, FALLBACK_SCALE
125
126 scale = float(cfg.get("force_scale", FALLBACK_SCALE))
127 g_attract = float(cfg.get("g_attract", FALLBACK_ATTRACT)) * scale
128 g_repel = float(cfg.get("g_repel", FALLBACK_REPEL)) * scale
129
130 rules: Dict[Tuple, float] = {}
131 for key, val in cfg.get("pair_rules", {}).items():
132 try:
133 i, j = (int(x) for x in key.split("-"))
134 internal_val = float(val) * scale
135 rules[(min(i, j), max(i, j))] = internal_val #max(-50.0, min(50.0, internal_val))
136 except (ValueError, TypeError):
137 print(f"[config] Skipping invalid pair_rules entry: {key!r}: {val!r}")
138
139 print(f"[config] Loaded from config.json — force_scale={scale}, "
140 f"g_attract={g_attract:.2f}, g_repel={g_repel:.2f}, "
141 f"{len(rules)} pair rule(s)")
142 return g_attract, g_repel, rules, scale

219 @staticmethod
220 def _init_obstacles(spec):
221 """@brief Resolve the @c obstacles config field into a list of (x, y, r) tuples.
222
223 Accepted forms: None/missing -> 4 random; int N -> N random; list of
224 [x, y, r] or {"x":..,"y":..,"r":..} -> exactly those. Malformed entries are
225 skipped with a warning.
226
227 @param spec The raw @c obstacles value from the config.
228 @return list of (x, y, radius) obstacle tuples.
229 """
230 def _random_set(n):
231 return [
232 (random.randint(150, SIM_WIDTH - 150),
233 random.randint(150, SIM_HEIGHT - 150),
234 random.randint(30, 70))
235 for _ in range(n)
236 ]
237
238 if spec is None:
239 print("[config] obstacles: random (4)")
240 return _random_set(4)
241
242 if isinstance(spec, int):
243 n = max(0, spec)
244 print(f"[config] obstacles: random ({n})")
245 return _random_set(n)
246
247 if not isinstance(spec, list):
248 print(f"[config] obstacles: invalid type {type(spec).__name__} — using random (4)")
249 return _random_set(4)
250
251 out = []
252 for idx, item in enumerate(spec):
253 try:
254 if isinstance(item, dict):
255 x, y, r = int(item["x"]), int(item["y"]), int(item["r"])
256 elif isinstance(item, (list, tuple)) and len(item) == 3:
257 x, y, r = int(item[0]), int(item[1]), int(item[2])
258 else:
259 raise ValueError("expected [x,y,r] or {x,y,r}")
260 if r <= 0:
261 raise ValueError("r must be positive")
262 out.append((x, y, r))
263 except (KeyError, ValueError, TypeError) as e:
264 print(f"[config] obstacles[{idx}] skipped: {item!r} ({e})")
265
266 print(f"[config] obstacles: explicit ({len(out)} from config.json)")
267 return out

271
272 Spawns one populated group per cluster when @c num_clusters is configured
273 (skipping the initial CFL round), or five randomly-assigned groups settled by

275 """

277 _nc = _cfg.get("num_clusters")

279 self.num_clusters = _nc

282 else:
283 self.num_clusters = random.randint(MIN_CLUSTERS, MAX_CLUSTERS)
284 _nc_configured = False
285 if _nc is not None:

289 (random.randint(D, SIM_WIDTH - D), random.randint(D, SIM_HEIGHT - D))

294 self.cluster_ages = {i: BLEND_MATURITY_ROUNDS for i in range(self.num_clusters)}

298 self._trail_counter = 0

300 self.cluster_colors = {i: CLUSTER_PALETTE[i] for i in range(self.num_clusters)}

379 def _sync_rules(self):
380 """@brief Add missing default rules for current cluster pairs and drop stale ones."""
381 current_keys = set()
382 for i in range(self.num_clusters):
383 for j in range(i, self.num_clusters):
384 key = (i, j)
385 current_keys.add(key)
386 if key not in self.rules:
387 self.rules[key] = self.g_attract if i == j else self.g_repel
388 for k in [k for k in self.rules if k not in current_keys]:
389 del self.rules[k]

429 def _trigger_explosion(self):
430 """@brief Scatter every agent and randomise its model — a re-convergence stress test."""
431 for p in self.all_particles:
432 p.vx = random.uniform(-80, 80)
433 p.vy = random.uniform(-80, 80)
434 p.x += random.uniform(-400, 400)
435 p.y += random.uniform(-200, 200)
436 p.model = np.array([
437 random.uniform(-1, 1), random.uniform(-1, 1),
438 0.1, 0.8, 0.0, 0.0, 1.0, 0.0,
439 ])
440 p.model[0:2] /= np.linalg.norm(p.model[0:2])
441 self.logger.log_explosion(self.cfl_round_counter)
442 print("\n!!! KA-BOOM !!! - Swarm disrupted.")

448 def _step(self, dt: float):
449 """@brief Advance the simulation by one physics tick.
450
451 Drifts each agent's inner target, runs local_train() for every agent,
452 fires an IFCA run_cfl_round() when the round timer elapses (handling
453 split/merge bookkeeping, metrics and logging), then applies
454 update_peer_alignment() and apply_physics_rules() and samples trails.
455
456 @param dt Integration timestep in seconds.
457 """
458 _DRIFT_SPEED = 0.10
459 _DRIFT_MARGIN = WALL_BOUNDARY * 4
460
461 # Drift each particle's own inner target independently
462 for p in self.all_particles:
463 p._target_angle += random.uniform(-0.01, 0.01)
464 nx = p._target_x + _DRIFT_SPEED * math.cos(p._target_angle)
465 ny = p._target_y + _DRIFT_SPEED * math.sin(p._target_angle)
466 if nx < _DRIFT_MARGIN or nx > SIM_WIDTH - _DRIFT_MARGIN:
467 p._target_angle = math.pi - p._target_angle
468 nx = max(_DRIFT_MARGIN, min(SIM_WIDTH - _DRIFT_MARGIN, nx))
469 if ny < _DRIFT_MARGIN or ny > SIM_HEIGHT - _DRIFT_MARGIN:
470 p._target_angle = -p._target_angle
471 ny = max(_DRIFT_MARGIN, min(SIM_HEIGHT - _DRIFT_MARGIN, ny))
472 p._target_x = nx
473 p._target_y = ny
474
475 # Recompute cluster targets as the average of each cluster's inner targets
476 sums: Dict[int, List[float]] = {}
477 counts: Dict[int, int] = {}
478 for p in self.all_particles:
479 cid = p.cluster_id if p.cluster_id >= 0 else p.target_idx
480 if cid not in sums:
481 sums[cid] = [0.0, 0.0]
482 counts[cid] = 0
483 sums[cid][0] += p._target_x
484 sums[cid][1] += p._target_y
485 counts[cid] += 1
486 for i in range(len(self.cluster_targets)):
487 if i in sums:
488 n = counts[i]
489 self.cluster_targets[i] = (sums[i][0] / n, sums[i][1] / n)
490
491 # Local training — each particle chases its own inner target
492 for p in self.all_particles:
493 local_train(p, (p._target_x, p._target_y), self.obstacles, learning_rate=0.02)
494
495 # Simulation round — fires every cluster_update_interval physics steps.
496 self.cluster_update_timer += 1
497 if self.cluster_update_timer >= self.cluster_update_interval:
498 self.cluster_update_timer = 0
499 self.cfl_round_counter += 1
500
501 if self.max_rounds is not None and self.cfl_round_counter >= self.max_rounds:
502 print(f"\n[SIM] max_rounds={self.max_rounds} reached — stopping simulation.")
503 self._stop_event.set()
504 pygame.event.post(pygame.event.Event(pygame.QUIT))
505 return
506
507 # Track metrics every round regardless of whether CFL is enabled,
508 # so the sparkline shows the contrast when the mode is toggled.
509 if self.all_particles:
510 _al = sum(p.model[6] for p in self.all_particles) / len(self.all_particles)
511 _ac = sum(p.model[2] for p in self.all_particles) / len(self.all_particles)
512 self._avg_loss_history.append(_al)
513 self._avg_conf_history.append(_ac)
514
515 if self.cfl_enabled:
516 transfers, self.kmeans, self.cluster_targets, self.cluster_colors, \
517 self.cluster_ages, self.num_clusters, event, self.cooldown_counter = run_cfl_round(
518 self.all_particles, self.kmeans, self.cluster_targets,
519 self.cluster_colors, self.cluster_ages, self.cooldown_counter,
520 )
521
522 max_idx = len(self.cluster_targets) - 1
523 for p in self.all_particles:
524 if p.cluster_id >= 0:
525 p.target_idx = min(p.cluster_id, max_idx)
526
527 event_type = event[0] if isinstance(event, tuple) else event
528 trail_maxlen = self.target_trails[0].maxlen if self.target_trails else FRAME_RATE * 3
529
530 if event_type == 'split':
531 self.target_trails.append(deque(maxlen=trail_maxlen))
532 self._sync_rules()
533 print(f" > SPLIT — now {self.num_clusters} clusters")
534 elif event_type == 'merge':
535 drop_idx = event[1]
536 if drop_idx < len(self.target_trails):
537 del self.target_trails[drop_idx]
538 elif self.target_trails:
539 self.target_trails.pop()
540 self._sync_rules()
541 print(f" > MERGE — now {self.num_clusters} clusters")
542
543 inertia = self.kmeans.inertia_ if hasattr(self.kmeans, 'inertia_') else 0.0
544 counts = {}
545 for p in self.all_particles:
546 counts[p.cluster_id] = counts.get(p.cluster_id, 0) + 1
547
548 print(f"\n[ROUND {self.cfl_round_counter}] CFL Complete")
549 print(f" > Inertia: {inertia:.2f}")
550 print(f" > Cluster Sizes: {dict(sorted(counts.items()))}")
551 if transfers:
552 print(f" > Migrations:")
553 for (old_id, new_id), cnt in sorted(transfers.items(), key=lambda x: -x[1]):
554 src = "Unassigned" if old_id == -1 else f"Cluster {old_id}"
555 print(f" - {cnt:3d} agents: {src} -> Cluster {new_id}")
556 else:
557 print(" > Migrations: (Stable)")
558 print("-" * 50)
559
560 self.logger.log_round(
561 round_num=self.cfl_round_counter,
562 particles=self.all_particles,
563 kmeans=self.kmeans,
564 cluster_targets=self.cluster_targets,

657 def run(self):
658 """@brief Thread entry point: drain commands, step, publish, and pace to PHYSICS_HZ until stopped."""
659 dt = 1.0 / FRAME_RATE # physics uses same dt the old code used per-step
660 period = 1.0 / PHYSICS_HZ
661
662 try:
663 while not self._stop_event.is_set():
664 t0 = time.perf_counter()
665
666 # Drain commands (all of them before the next physics step)
667 while True:
668 try:
669 self._handle_cmd(self.cmd_queue.get_nowait())
670 except queue.Empty:
671 break
672
673 self._step(dt)
674
675 snap = self._build_snapshot()
676 with self._snapshot_lock:
677 self._snapshot = snap
678
679 elapsed = time.perf_counter() - t0
680 remaining = period - elapsed
681 if remaining > 0.0005:
682 time.sleep(remaining)
683 finally:
684 self.logger.close()

687 """@brief Signal the thread to stop after the current step (triggers logger teardown)."""
688 self._stop_event.set()

1043 if rect.collidepoint(event.pos):
1044 hit = (min(i, j), max(i, j))
1045 break
1046
1047 if hit is not None:
1048 if self.last_click_cell == hit and now - self.last_click_time < 400:
1049 i, j = hit

1051 self.editing_cell = None
1052 self.edit_buffer = ""
1053 self.last_click_cell = None
1054 else:
1055 self._commit_edit(snap)

1058 self.last_click_cell = hit

1060 else:
