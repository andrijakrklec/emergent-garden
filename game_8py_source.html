51def _load_merged_config(config_path: str) -> dict:
52 """@brief Resolve the layered configuration: master + emergent preset + per-run override.
53
54 Three layers are merged, each overriding the previous:
55 1. the master config.json (globals plus the @c emergent_preset selector);
56 2. the emergent preset it names (configs/emergent/<name>.json), if any;
57 3. the per-run --config file (used by run_all.py), if different from master.
58 The @c pair_rules table is deep-merged at each step; comment keys (those
59 starting with '_') are dropped.
60
61 @param config_path Path to the per-run config (may equal the master path).
62 @return dict the fully merged configuration.
63 """
64 def _read(path: str) -> dict:
65 try:
66 with open(path, "r") as f:
67 raw = json.load(f)
68 return {k: v for k, v in raw.items() if not k.startswith("_")}
69 except (FileNotFoundError, json.JSONDecodeError):
70 return {}
71
72 def _deep_merge(base: dict, override: dict) -> dict:
73 """Shallow-merge two configs, but deep-merge their pair_rules tables."""
74 merged = {**base, **override}
75 base_rules = base.get("pair_rules", {})
76 ovr_rules = override.get("pair_rules", {})
77 if base_rules or ovr_rules:
78 merged["pair_rules"] = {**base_rules, **ovr_rules}
79 return merged
80
81 cfg = _read(DEFAULT_CONFIG_PATH)
82
83 # Overlay the named emergent preset so switching `emergent_preset` in
84 # config.json swaps the whole attract/repel personality in one place.
85 preset_name = cfg.get("emergent_preset")
86 if preset_name:
87 preset_path = os.path.join(
88 os.path.dirname(DEFAULT_CONFIG_PATH), "configs", "emergent", f"{preset_name}.json"
89 )
90 preset = _read(preset_path)
91 if preset:
92 cfg = _deep_merge(cfg, preset)
93 else:
94 print(f"[config] emergent_preset {preset_name!r} not found at {preset_path} — ignoring.")
95
96 # Overlay the per-run --config file (run_all.py ablation), if any.
97 if os.path.abspath(config_path) == os.path.abspath(DEFAULT_CONFIG_PATH):
98 return cfg
99 override = _read(config_path)
100 if not override:
101 return cfg
102 return _deep_merge(cfg, override)

105def _load_force_config(config_path: str) -> Tuple[float, float, Dict, float]:
106 """@brief Load g_attract, g_repel and the optional pair_rules from merged config.
107
108 Config force values are in a readable ~-1..+1 scale; @c force_scale multiplies
109 them into internal units before use. Falls back to hardcoded defaults if the
110 config is missing or malformed.
111
112 @param config_path Path to the per-experiment config.
113 @return tuple (g_attract, g_repel, rules, force_scale) where @c rules maps (i, j) -> float.
114 """
115 FALLBACK_SCALE = 25.0
116 FALLBACK_ATTRACT = -10.0 / FALLBACK_SCALE # = -0.4
117 FALLBACK_REPEL = 25.0 / FALLBACK_SCALE # = 1.0
118
119 cfg = _load_merged_config(config_path)
120 if not cfg:
121 print("[config] config.json not found — using hardcoded defaults.")
122 return -10.0, 25.0, {}, FALLBACK_SCALE
123
124 scale = float(cfg.get("force_scale", FALLBACK_SCALE))
125 g_attract = float(cfg.get("g_attract", FALLBACK_ATTRACT)) * scale
126 g_repel = float(cfg.get("g_repel", FALLBACK_REPEL)) * scale
127
128 rules: Dict[Tuple, float] = {}
129 for key, val in cfg.get("pair_rules", {}).items():
130 try:
131 i, j = (int(x) for x in key.split("-"))
132 internal_val = float(val) * scale
133 rules[(min(i, j), max(i, j))] = internal_val #max(-50.0, min(50.0, internal_val))
134 except (ValueError, TypeError):
135 print(f"[config] Skipping invalid pair_rules entry: {key!r}: {val!r}")
136
137 print(f"[config] Loaded from config.json — force_scale={scale}, "
138 f"g_attract={g_attract:.2f}, g_repel={g_repel:.2f}, "
139 f"{len(rules)} pair rule(s)")
140 return g_attract, g_repel, rules, scale

211 def _init_obstacles(spec):
212 """@brief Resolve the @c obstacles config field into a list of (x, y, r) tuples.
213
214 Accepted forms: None/missing -> 4 random; int N -> N random; list of
215 [x, y, r] or {"x":..,"y":..,"r":..} -> exactly those. Malformed entries are
216 skipped with a warning.
217
218 @param spec The raw @c obstacles value from the config.
219 @return list of (x, y, radius) obstacle tuples.
220 """
221 def _random_set(n):
222 return [
223 (random.randint(150, SIM_WIDTH - 150),
224 random.randint(150, SIM_HEIGHT - 150),
225 random.randint(30, 70))
226 for _ in range(n)
227 ]
228
229 if spec is None:
230 print("[config] obstacles: random (4)")
231 return _random_set(4)
232
233 if isinstance(spec, int):
234 n = max(0, spec)
235 print(f"[config] obstacles: random ({n})")
236 return _random_set(n)
237
238 if not isinstance(spec, list):
239 print(f"[config] obstacles: invalid type {type(spec).__name__} — using random (4)")
240 return _random_set(4)
241
242 out = []
243 for idx, item in enumerate(spec):
244 try:
245 if isinstance(item, dict):
246 x, y, r = int(item["x"]), int(item["y"]), int(item["r"])
247 elif isinstance(item, (list, tuple)) and len(item) == 3:
248 x, y, r = int(item[0]), int(item[1]), int(item[2])
249 else:
250 raise ValueError("expected [x,y,r] or {x,y,r}")
251 if r <= 0:
252 raise ValueError("r must be positive")
253 out.append((x, y, r))
254 except (KeyError, ValueError, TypeError) as e:
255 print(f"[config] obstacles[{idx}] skipped: {item!r} ({e})")
256
257 print(f"[config] obstacles: explicit ({len(out)} from config.json)")
258 return out

369 def _sync_rules(self):
370 """@brief Add missing default rules for current cluster pairs and drop stale ones."""
371 current_keys = set()
372 for i in range(self.num_clusters):
373 for j in range(i, self.num_clusters):
374 key = (i, j)
375 current_keys.add(key)
376 if key not in self.rules:
377 self.rules[key] = self.g_attract if i == j else self.g_repel
378 for k in [k for k in self.rules if k not in current_keys]:
379 del self.rules[k]

419 def _trigger_explosion(self):
420 """@brief Scatter every agent and randomise its model — a re-convergence stress test."""
421 for p in self.all_particles:
422 p.vx = random.uniform(-80, 80)
423 p.vy = random.uniform(-80, 80)
424 p.x += random.uniform(-400, 400)
425 p.y += random.uniform(-200, 200)
426 p.model = np.array([
427 random.uniform(-1, 1), random.uniform(-1, 1),

430 p.model[0:2] /= np.linalg.norm(p.model[0:2])

438 def _step(self, dt: float):
439 """@brief Advance the simulation by one physics tick.
440
441 Drifts each agent's inner target, runs local_train() for every agent,
442 fires an IFCA run_cfl_round() when the round timer elapses (handling
443 split/merge bookkeeping, metrics and logging), then applies
444 update_peer_alignment() and apply_physics_rules() and samples trails.
445
446 @param dt Integration timestep in seconds.
447 """
448 _DRIFT_SPEED = 0.10
449 _DRIFT_MARGIN = WALL_BOUNDARY * 4
450
451 # Drift each particle's own inner target independently
452 for p in self.all_particles:
453 p._target_angle += random.uniform(-0.01, 0.01)
454 nx = p._target_x + _DRIFT_SPEED * math.cos(p._target_angle)
455 ny = p._target_y + _DRIFT_SPEED * math.sin(p._target_angle)
456 if nx < _DRIFT_MARGIN or nx > SIM_WIDTH - _DRIFT_MARGIN:
457 p._target_angle = math.pi - p._target_angle
458 nx = max(_DRIFT_MARGIN, min(SIM_WIDTH - _DRIFT_MARGIN, nx))
459 if ny < _DRIFT_MARGIN or ny > SIM_HEIGHT - _DRIFT_MARGIN:
460 p._target_angle = -p._target_angle
461 ny = max(_DRIFT_MARGIN, min(SIM_HEIGHT - _DRIFT_MARGIN, ny))
462 p._target_x = nx
463 p._target_y = ny
464
465 # Recompute cluster targets as the average of each cluster's inner targets
466 sums: Dict[int, List[float]] = {}
467 counts: Dict[int, int] = {}
468 for p in self.all_particles:
469 cid = p.cluster_id if p.cluster_id >= 0 else p.target_idx
470 if cid not in sums:
471 sums[cid] = [0.0, 0.0]
472 counts[cid] = 0
473 sums[cid][0] += p._target_x
474 sums[cid][1] += p._target_y
475 counts[cid] += 1
476 for i in range(len(self.cluster_targets)):
477 if i in sums:
478 n = counts[i]
479 self.cluster_targets[i] = (sums[i][0] / n, sums[i][1] / n)
480
481 # Local training — each particle chases its own inner target
482 for p in self.all_particles:
483 local_train(p, (p._target_x, p._target_y), self.obstacles, learning_rate=0.02)
484
485 # Simulation round — fires every cluster_update_interval physics steps.
486 self.cluster_update_timer += 1
487 if self.cluster_update_timer >= self.cluster_update_interval:
488 self.cluster_update_timer = 0
489 self.cfl_round_counter += 1
490
491 if self.max_rounds is not None and self.cfl_round_counter >= self.max_rounds:
492 print(f"\n[SIM] max_rounds={self.max_rounds} reached — stopping simulation.")
493 self._stop_event.set()
494 pygame.event.post(pygame.event.Event(pygame.QUIT))
495 return
496
497 # Track metrics every round regardless of whether CFL is enabled,
498 # so the sparkline shows the contrast when the mode is toggled.
499 if self.all_particles:
500 _al = sum(p.model[6] for p in self.all_particles) / len(self.all_particles)
501 _ac = sum(p.model[2] for p in self.all_particles) / len(self.all_particles)
502 self._avg_loss_history.append(_al)
503 self._avg_conf_history.append(_ac)
504
505 if self.cfl_enabled:
506 transfers, self.kmeans, self.cluster_targets, self.cluster_colors, \
507 self.cluster_ages, self.num_clusters, event, self.cooldown_counter = run_cfl_round(
508 self.all_particles, self.kmeans, self.cluster_targets,
509 self.cluster_colors, self.cluster_ages, self.cooldown_counter,
510 )
511
512 max_idx = len(self.cluster_targets) - 1
513 for p in self.all_particles:
514 if p.cluster_id >= 0:
515 p.target_idx = min(p.cluster_id, max_idx)
516
517 event_type = event[0] if isinstance(event, tuple) else event
518 trail_maxlen = self.target_trails[0].maxlen if self.target_trails else FRAME_RATE * 3
519
520 if event_type == 'split':
521 self.target_trails.append(deque(maxlen=trail_maxlen))
522 self._sync_rules()
523 print(f" > SPLIT — now {self.num_clusters} clusters")
524 elif event_type == 'merge':
525 drop_idx = event[1]
526 if drop_idx < len(self.target_trails):
527 del self.target_trails[drop_idx]
528 elif self.target_trails:
529 self.target_trails.pop()
530 self._sync_rules()
531 print(f" > MERGE — now {self.num_clusters} clusters")
532
533 inertia = self.kmeans.inertia_ if hasattr(self.kmeans, 'inertia_') else 0.0
534 counts = {}
535 for p in self.all_particles:
536 counts[p.cluster_id] = counts.get(p.cluster_id, 0) + 1
537
538 print(f"\n[ROUND {self.cfl_round_counter}] CFL Complete")
539 print(f" > Inertia: {inertia:.2f}")
540 print(f" > Cluster Sizes: {dict(sorted(counts.items()))}")
541 if transfers:
542 print(f" > Migrations:")
543 for (old_id, new_id), cnt in sorted(transfers.items(), key=lambda x: -x[1]):
544 src = "Unassigned" if old_id == -1 else f"Cluster {old_id}"
545 print(f" - {cnt:3d} agents: {src} -> Cluster {new_id}")
546 else:
547 print(" > Migrations: (Stable)")
548 print("-" * 50)
549
550 self.logger.log_round(
551 round_num=self.cfl_round_counter,
552 particles=self.all_particles,
553 kmeans=self.kmeans,
554 cluster_targets=self.cluster_targets,
555 transfers=transfers,
556 event=event_type,
557 num_clusters=self.num_clusters,
558 )
559 else:
560 print(f"\n[ROUND {self.cfl_round_counter}] (CFL disabled — individual learning only)")
561 self.logger.log_round(
562 round_num=self.cfl_round_counter,
563 particles=self.all_particles,
564 kmeans=self.kmeans,
565 cluster_targets=self.cluster_targets,
566 transfers={},
567 event=None,
568 num_clusters=self.num_clusters,
569 )
570
571 if self.cfl_round_counter % 20 == 0:
572 self.logger.plot_all()
573
574 update_peer_alignment(self.all_particles)
575 apply_physics_rules(

577 dt, self.rules, attraction_enabled=self.attraction_enabled,

585 maxlen = self.target_trails[0].maxlen if self.target_trails else FRAME_RATE * 3
586 self.target_trails.append(deque(maxlen=maxlen))

600 """

603 for p in self.all_particles
604 ]
605 particle_target_indices = [(p._target_x, p._target_y) for p in self.all_particles]
606 stats: Dict[int, int] = {}
607 for p in self.all_particles:
608 stats[p.cluster_id] = stats.get(p.cluster_id, 0) + 1
609
610 if self.all_particles:
611 avg_loss = sum(p.model[6] for p in self.all_particles) / len(self.all_particles)
612 avg_conf = sum(p.model[2] for p in self.all_particles) / len(self.all_particles)
613 else:
614 avg_loss = avg_conf = 0.0
615
616 return SimSnapshot(
617 particles=particle_data,
618 particle_inner_targets=particle_target_indices,
619 cluster_targets=list(self.cluster_targets),
620 cluster_colors=dict(self.cluster_colors),
621 obstacles=list(self.obstacles),
622 target_trails=[list(t) for t in self.target_trails],
623 cfl_round_counter=self.cfl_round_counter,
624 num_clusters=self.num_clusters,
625 cluster_stats=stats,
626 g_attract=self.g_attract,
627 g_repel=self.g_repel,
628 force_scale=self.force_scale,
629 rules=dict(self.rules),
630 cfl_enabled=self.cfl_enabled,
631 attraction_enabled=self.attraction_enabled,

633 avg_confidence=avg_conf,

635 conf_history=list(self._avg_conf_history),

644 # Thread entry point
645 # ------------------------------------------------------------------
646
647 def run(self):
648 """@brief Thread entry point: drain commands, step, publish, and pace to PHYSICS_HZ until stopped."""
649 dt = 1.0 / FRAME_RATE # physics uses same dt the old code used per-step
650 period = 1.0 / PHYSICS_HZ
651
652 try:
653 while not self._stop_event.is_set():
654 t0 = time.perf_counter()
655
656 # Drain commands (all of them before the next physics step)
657 while True:
658 try:
659 self._handle_cmd(self.cmd_queue.get_nowait())
660 except queue.Empty:
661 break
662
663 self._step(dt)
664
665 snap = self._build_snapshot()
666 with self._snapshot_lock:
667 self._snapshot = snap
668
669 elapsed = time.perf_counter() - t0
670 remaining = period - elapsed
671 if remaining > 0.0005:
672 time.sleep(remaining)
673 finally:
674 self.logger.close()

676 def stop(self):
677 """@brief Signal the thread to stop after the current step (triggers logger teardown)."""
678 self._stop_event.set()

1008 if event.type == pygame.QUIT:
1009 self.game_running = False
1010
1011 elif event.type == pygame.MOUSEBUTTONDOWN and event.button == 1:
1012 if self.bomb_rect.collidepoint(event.pos):
1013 self._commit_edit(snap)
1014 self.cmd_queue.put({'type': 'detonate'})
1015 elif self.cfl_toggle_rect.collidepoint(event.pos):
1016 self._commit_edit(snap)
1017 self.cmd_queue.put({'type': 'toggle_cfl'})
1018 elif self.attraction_toggle_rect.collidepoint(event.pos):
1019 self._commit_edit(snap)
1020 self.cmd_queue.put({'type': 'toggle_attraction'})
1021 else:

1023 hit = None
1024 for (i, j), rect in self.rule_cell_rects.items():
1025 if rect.collidepoint(event.pos):
1026 hit = (min(i, j), max(i, j))
1027 break
1028
1029 if hit is not None:
1030 if self.last_click_cell == hit and now - self.last_click_time < 400:
1031 i, j = hit
1032 self.cmd_queue.put({'type': 'reset_rule', 'i': i, 'j': j})
1033 self.editing_cell = None
1034 self.edit_buffer = ""
1035 self.last_click_cell = None
1036 else:
1037 self._commit_edit(snap)
1038 self.editing_cell = hit
1039 self.edit_buffer = f"{self._current_rule_val(hit, snap) / (snap.force_scale or 1.0):.2f}"
1040 self.last_click_cell = hit
1041 self.last_click_time = now
1042 else:
1043 self._commit_edit(snap)
1044 self.last_click_cell = None
1045
1046 elif event.type == pygame.KEYDOWN:
1047 if event.key == pygame.K_t and self.editing_cell is None:
1048 self.show_target_lines = not self.show_target_lines
1049 elif self.editing_cell is not None:
1050 if event.key in (pygame.K_RETURN, pygame.K_KP_ENTER):
1051 self._commit_edit(snap)
1052 elif event.key == pygame.K_ESCAPE:
1053 self.editing_cell = None
1054 self.edit_buffer = ""

1057 elif event.unicode in "0123456789":
1058 self.edit_buffer += event.unicode
1059 elif event.unicode == "-" and self.edit_buffer == "":
1060 self.edit_buffer = "-"
1061 elif event.unicode == "." and "." not in self.edit_buffer:
1062 self.edit_buffer += "."
