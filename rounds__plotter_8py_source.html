1"""@file rounds_plotter.py
2@brief Standalone, fully-customizable re-plotter for a SimLogger ``rounds.csv``.
3
4Reads a single ``rounds.csv`` (written by SimLogger) and
5renders the same round-level metrics that land in ``global_metrics.png`` — but
6each metric becomes its **own** figure instead of a 2x3 panel, and every cosmetic
7property is overridable: titles, axis labels, axis tick granularity, colors,
8y-limits, theme, figure size, DPI, fonts, event markers and output format.
9
10Unlike SimLogger (which plots from its in-memory history during a live run), this
11module is post-hoc: point it at any ``rounds.csv`` on disk and re-style the plots
12for a thesis figure without re-running the simulation.
13
14Two ways to use it.
15
16CLI
17---
18 # default 6 metrics, one PNG each, dark theme, alongside the CSV in ./plots
19 python -m src.utils.rounds_plotter logs/run_.../rounds.csv
20
21 # rename axes / title, set tick granularity, light theme, save as PDF
22 python -m src.utils.rounds_plotter logs/run_.../rounds.csv \
23 --metrics avg_local_loss,avg_direction_precision \
24 --title avg_local_loss="Avg local loss" \
25 --ylabel avg_local_loss="loss function" \
26 --xlabel "round" --xstep 5 --ystep avg_local_loss=0.05 \
27 --theme light --format pdf --out figures/
28
29 python -m src.utils.rounds_plotter rounds.csv --all # every column in the CSV
30
31 # comparison: overlay several runs (folders ok), one line each
32 python -m src.utils.rounds_plotter \
33 logs/cfl_on logs/cfl_off \
34 --labels "CFL on,CFL off" --colors "#5599dd,#e05555" \
35 --metrics avg_local_loss,avg_direction_precision --out figures/
36
37 # average several runs into ONE mean line (no comparison)
38 python -m src.utils.rounds_plotter \
39 logs/cells/cfl_on__emergent_on_run01 logs/cells/cfl_on__emergent_on_run02 \
40 --avg --labels "CFL on (mean)" --metrics avg_local_loss --out figures/
41
42 # averaged-group comparison: average each cell's repeated runs into one line
43 python -m src.utils.rounds_plotter \
44 --group "CFL on=logs/cells/cfl_on__emergent_on_run01,logs/cells/cfl_on__emergent_on_run02" \
45 --group "CFL off=logs/cells/cfl_off__emergent_on_run01,logs/cells/cfl_off__emergent_on_run02" \
46 --colors "#5599dd,#e05555" --metrics avg_local_loss --out figures/

251def _to_float(raw: object) -> float:
252 """@brief Parse a CSV cell to float; blank / unparsable -> NaN.
253
254 SimLogger writes empty strings for metrics that are N/A in a given run.
255 Those become NaN here and are filtered out per-series, exactly as SimLogger
256 filters @c None.
257 """
258 if raw is None:
259 return float("nan")
260 s = str(raw).strip()
261 if s == "":
262 return float("nan")
263 try:
264 return float(s)
265 except ValueError:
266 return float("nan")

290# Markdown-style **bold** spans -> matplotlib mathtext. Text outside the spans is
291# left as ordinary text (its font/diacritics untouched); only the bold span is
292# rendered via mathtext, so it stays ASCII-friendly. No "**" -> returned as-is.
293_BOLD_RE = re.compile(r"\*\*(.+?)\*\*", re.DOTALL)
294
295
296def _bold_markup(text: Optional[str]) -> Optional[str]:
297 """@brief Convert ``**bold**`` spans in a label/title to mathtext bold.
298
299 Inside a span, characters that are special to mathtext are escaped and spaces
300 are turned into ``\\ `` (mathtext collapses raw spaces). Use for ASCII text;
301 accented letters inside a bold span may not render in every matplotlib build.
302
303 @param text Any label/title string (or None).
304 @return The string with bold spans converted, or the original if it has none.
305 """
306 if not text or "**" not in text:
307 return text
308
309 def _sub(m: "re.Match") -> str:
310 inner = m.group(1)
311 for ch in "\\_^${}#%": # backslash first
312 inner = inner.replace(ch, "\\" + ch)
313 inner = inner.replace(" ", r"\ ")
314 return r"$\mathbf{" + inner + r"}$"
315
316 return _BOLD_RE.sub(_sub, text)

326 def __init__(
327 self,
328 csv_path: str,
329 style: Optional[PlotStyle] = None,
330 registry: Optional[Dict[str, MetricSpec]] = None,
331 ) -> None:
332 """@brief Read and parse the CSV up front.
333
334 @param csv_path Path to a SimLogger ``rounds.csv`` (or the folder that
335 contains it — a directory is resolved to ``<dir>/rounds.csv``).
336 @param style Figure-wide @ref PlotStyle (defaults applied if omitted).
337 @param registry Column -> @ref MetricSpec map (defaults to the built-in set).
338 """
339 self.csv_path = _resolve_csv(csv_path)
340 csv_path = self.csv_path
341 self.style = style or PlotStyle()
342 self.registry = registry if registry is not None else default_metric_registry()
343 self.rounds, self.data, self.events = self._load(csv_path)

345 # ── loading ────────────────────────────────────────────────────────────────
346
347 @staticmethod
348 def _load(csv_path: str) -> Tuple[List[int], Dict[str, List[float]], List[Tuple[int, str]]]:
349 """@brief Parse rounds.csv into x-rounds, per-column float series, and events.
350
351 @return (rounds, data, events) where @c data maps each numeric column to a
352 list aligned with @c rounds (NaN where blank), and @c events is a list
353 of (round, type) drawn from the per-row ``event`` column.
354 """
355 # utf-8-sig tolerates a BOM (Excel / some editors add one), which would
356 # otherwise corrupt the first header name ("round" -> "﻿round").
357 with open(csv_path, newline="", encoding="utf-8-sig") as fh:
358 reader = csv.DictReader(fh)
359 fieldnames = reader.fieldnames or []
360 value_cols = [f for f in fieldnames if f not in ("round", "event")]
361 rounds: List[int] = []
362 data: Dict[str, List[float]] = {c: [] for c in value_cols}
363 events: List[Tuple[int, str]] = []
364 for row in reader:
365 try:
366 rnd = int(float(row["round"]))
367 except (KeyError, TypeError, ValueError):
368 continue
369 rounds.append(rnd)
370 for col in value_cols:
371 data[col].append(_to_float(row.get(col, "")))
372 ev = (row.get("event") or "").strip().lower()
373 if ev in EVENT_COLORS:
374 events.append((rnd, ev))
375
376 # Derived column: migrations normalized by swarm size (fraction of
377 # particles that migrated that round) — comparable across runs/cells of
378 # different particle counts. NaN where inputs are missing or N == 0.
379 if "total_migrations" in data and "num_particles" in data:
380 data["migration_rate"] = [
381 (m / n) if (not _is_nan(m) and not _is_nan(n) and n) else float("nan")
382 for m, n in zip(data["total_migrations"], data["num_particles"])
383 ]
384 return rounds, data, events
385
386 def _series(self, column: str) -> Tuple[List[int], List[float]]:
387 """@brief (rounds, values) for a column with NaN rows dropped.
388
389 @throws KeyError if the column is absent from the CSV.
390 """
391 if column not in self.data:
392 raise KeyError(
393 f"Column {column!r} not in {self.csv_path} "
394 f"(available: {', '.join(sorted(self.data))})"
395 )
396 xs, ys = [], []
397 for r, v in zip(self.rounds, self.data[column]):
398 if _is_nan(v):
399 continue
400 xs.append(r)
401 ys.append(v)
402 return xs, ys

406 def _plt(self):
407 """@brief Import pyplot, forcing the Agg backend unless an interactive show was requested."""
408 import matplotlib
409 if not self.style.show:
410 matplotlib.use("Agg")
411 import matplotlib.pyplot as plt
412 return plt

414 def _theme(self) -> Dict[str, str]:
415 return THEMES.get(self.style.theme, THEMES["dark"])
416
417 def _style_axes(self, fig, ax) -> None:
418 s, t = self.style, self._theme()
419 fig.patch.set_facecolor(t["fig_bg"])
420 ax.set_facecolor(t["ax_bg"])
421 ax.tick_params(colors=t["fg"], labelsize=s.tick_fontsize)
422 ax.xaxis.label.set_color(t["fg"])
423 ax.yaxis.label.set_color(t["fg"])
424 ax.title.set_color(t["fg"])
425 for spine in ax.spines.values():
426 spine.set_edgecolor(t["spine"])
427 if s.grid:
428 ax.grid(True, color=t["grid"], alpha=s.grid_alpha, linewidth=0.7)
429 ax.set_axisbelow(True)
430
431 def _mark_events(self, ax) -> List[str]:
432 """@brief Draw split/merge vertical markers; return the event types present."""
433 present: List[str] = []
434 round_set = set(self.rounds)
435 for rnd, ev in self.events:
436 if rnd not in round_set:
437 continue
438 ax.axvline(rnd, color=EVENT_COLORS[ev], linewidth=1.2,
439 linestyle="--", alpha=0.7)
440 if ev not in present:
441 present.append(ev)
442 return present
443
444 @staticmethod
445 def _event_patches(present: List[str]):
446 from matplotlib.lines import Line2D
447 return [
448 Line2D([0], [0], color=EVENT_COLORS[ev], lw=1.5, ls="--", label=ev)
449 for ev in present
450 ]

460 def plot_metric(self, spec: MetricSpec, out_dir: str) -> Optional[str]:
461 """@brief Render and save one metric as its own figure.
462
463 @param spec The @ref MetricSpec describing column + cosmetics.
464 @param out_dir Directory to write into (created if missing).
465 @return Saved file path, or @c None if the column has no plottable data.
466 """
467 s = self.style
468 t = self._theme()
469 try:
470 xs, ys = self._series(spec.column)
471 except KeyError as exc:
472 print(f" [skip] {spec.column}: {exc}")
473 return None
474 if not xs:
475 print(f" [skip] {spec.column}: no data (all blank/NaN)")
476 return None
477
478 plt = self._plt()
479 from matplotlib.ticker import MaxNLocator, MultipleLocator
480
481 fig, ax = plt.subplots(figsize=s.figsize)
482 self._style_axes(fig, ax)
483
484 line, = ax.plot(
485 xs, ys, color=spec.color, linewidth=s.linewidth,
486 marker=s.marker, markersize=s.markersize, label=spec.title,
487 )
488
489 present = self._mark_events(ax) if s.mark_events else []
490
491 ax.set_xlabel(_bold_markup(spec.xlabel or s.xlabel), fontsize=s.label_fontsize, fontweight=s.label_weight)
492 ax.set_ylabel(_bold_markup(spec.ylabel), fontsize=s.label_fontsize, fontweight=s.label_weight)
493 ax.set_title(_bold_markup(spec.title), fontsize=s.title_fontsize, fontweight="bold")
494
495 if spec.ylim is not None:
496 ax.set_ylim(*spec.ylim)
497 if spec.integer_y:
498 ax.yaxis.set_major_locator(MaxNLocator(integer=True))
499 elif spec.y_step:
500 ax.yaxis.set_major_locator(MultipleLocator(spec.y_step))
501 if s.x_step:
502 ax.xaxis.set_major_locator(MultipleLocator(s.x_step))
503
504 if s.suptitle:
505 fig.suptitle(_bold_markup(s.suptitle), color=t["fg"], fontsize=s.title_fontsize - 1)
506
507 if s.show_legend:
508 ax.legend(
509 handles=[line] + self._event_patches(present),
510 fontsize=s.legend_fontsize, facecolor=t["legend_bg"],
511 labelcolor=t["fg"], edgecolor=t["spine"],
512 )
513
514 fig.tight_layout()
515 os.makedirs(out_dir, exist_ok=True)
516 stem = spec.filename or spec.column
517 path = os.path.join(out_dir, f"{stem}.{s.output_format}")
518 fig.savefig(path, dpi=s.dpi, bbox_inches="tight", facecolor=fig.get_facecolor())
519 if s.show:
520 plt.show()
521 plt.close(fig)
522 print(f" [ok] {os.path.basename(path)}")
523 if s.export_csv:
524 csv_path = os.path.join(out_dir, f"{stem}.csv")
525 _write_data_csv(csv_path, ["round", spec.column], list(zip(xs, ys)))
526 print(f" [ok] {os.path.basename(csv_path)}")
527 return path

530 self,
531 columns: Optional[List[str]] = None,
532 out_dir: Optional[str] = None,
533 ) -> List[str]:
534 """@brief Render the chosen columns, one figure each.
535
536 @param columns Columns to plot; defaults to the six SimLogger global
537 metrics (those present in this CSV).
538 @param out_dir Output directory; defaults to ``<csv_dir>/plots``.
539 @return List of saved file paths.
540 """
541 if columns is None:
542 columns = [c for c in DEFAULT_METRIC_COLUMNS if c in self.data]

552 saved.append(path)

568class ComparisonPlotter:
569 """@brief Overlay one metric from several runs / averaged groups on one figure.
570
571 One figure is produced per metric; within it, each *group* is a separate
572 labeled line. A group is one or more run folders: with several folders their
573 per-round metrics are **averaged** (NaN-aware, aligned by round number), so a
574 cell of repeated runs collapses to a single mean line. Built for cross-cell
575 comparison (e.g. the CFL on/off × emergent on/off ablation, each repeated N
576 times). Cosmetics reuse the same @ref MetricSpec (title, y-label, y-limits,
577 tick granularity) and @ref PlotStyle as @ref RoundsPlotter; line color is one
578 per group. Event markers are not drawn (ambiguous across runs). Output files
579 are prefixed ``compare_``.
580 """

588 ) -> None:

591 @param groups Iterable of group specs. Each group is a dict
592 ``{"label":.., "paths":[..], "color":..}`` or a tuple
593 ``(label, paths)`` / ``(label, paths, color)`` where @c paths is a
594 list of run folders (or ``rounds.csv`` paths). A single path may be
595 given instead of a list. A blank label defaults to the common run
596 folder name.
597 @param style Shared @ref PlotStyle.
598 @param registry Shared column -> @ref MetricSpec map.
599 @param palette Per-group color cycle (defaults to @ref SERIES_PALETTE).
600 """
601 self.style = style or PlotStyle()
602 self.registry = registry if registry is not None else default_metric_registry()
603 self.palette = palette or SERIES_PALETTE
604 self.series: List[Dict[str, Any]] = []
605 for g in groups:
606 if isinstance(g, dict):
607 label, paths, color = g.get("label"), g.get("paths"), g.get("color")
608 elif isinstance(g, (tuple, list)):
609 label = g[0]
610 paths = g[1] if len(g) > 1 else None
611 color = g[2] if len(g) > 2 else None
612 else: # a bare path string
613 label, paths, color = None, g, None
614 if isinstance(paths, str):
615 paths = [paths]
616 plotters = [RoundsPlotter(p, self.style, self.registry) for p in (paths or [])]
617 self.series.append({
618 "label": label or self._auto_label(plotters),
619 "color": color,
620 "plotters": plotters,
621 })

623 @staticmethod
624 def _auto_label(plotters: List["RoundsPlotter"]) -> str:
625 """@brief Default label: the run folder name (single) or its base (group)."""
626 if not plotters:
627 return "(empty)"
628 names = [os.path.basename(os.path.dirname(os.path.abspath(pl.csv_path)))
629 for pl in plotters]
630 if len(names) == 1:
631 return names[0] or os.path.basename(plotters[0].csv_path)
632 # several runs: drop a trailing run-number suffix and show the shared base
633 base = re.sub(r"[._-]?run[\s_-]*\d+$", "", names[0], flags=re.IGNORECASE)
634 return f"{base or names[0]} (mean of {len(names)})"

636 @staticmethod
637 def _avg_series(plotters: List["RoundsPlotter"], column: str):
638 """@brief Per-round mean (and population std) of a column across runs.
639
640 Aligns by round number; each round is averaged over the runs that have a
641 (non-NaN) value there. @return (xs, means, stds).
642 """
643 from collections import defaultdict
644 buckets: Dict[int, List[float]] = defaultdict(list)
645 for pl in plotters:
646 try:
647 xs, ys = pl._series(column)
648 except KeyError:
649 continue
650 for x, y in zip(xs, ys):
651 buckets[x].append(y)
652 xs = sorted(buckets)
653 means, stds = [], []
654 for x in xs:
655 vals = buckets[x]
656 m = sum(vals) / len(vals)
657 means.append(m)
658 var = sum((v - m) ** 2 for v in vals) / len(vals)
659 stds.append(var ** 0.5)
660 return xs, means, stds
