10# --- CLUSTER-COUNT BOUNDS ---
11_MIN_CLUSTERS = 2
12_MAX_CLUSTERS = 6

14# Split fires when EITHER (a) avg_loss exceeds threshold (a cluster genuinely
15# can't serve its members) OR (b) directional coherence among members drops
16# below threshold (members disagree on heading — the natural IFCA split signal,
17# since IFCA already pins avg_loss near the local minimum).
18_SPLIT_LOSS_THRESHOLD = 0.30
19_SPLIT_COHERENCE_THRESHOLD = 0.70 # mean cos-sim to cluster-mean direction
20
21# Merge thresholds: cosine sim of mean direction (loosened from 0.97 — IFCA's
22# bias-averaged means cluster more cleanly so 0.92 is a reasonable bar) plus
23# spatial-target proximity (two clusters whose targets have drifted close
24# together are functionally redundant under IFCA's loss).
25_MERGE_SIMILARITY_THRESHOLD = 0.92
26_MERGE_TARGET_DISTANCE = 120.0 # px — targets closer than this favor merge
27_MIN_CLUSTER_SIZE = 3 # clusters smaller than this get absorbed
28
29_BLEND_MATURITY_ROUNDS = 15 # rounds until a new cluster reaches full global blend
30_BLEND_LOCAL_NEW = 0.75 # local weight for a freshly split cluster
31
32# Non-IID bias structure: each particle's _local_bias direction is a mix of
33# its true cluster's shared bias direction and a per-particle random direction.
34# BIAS_CLUSTER_FRACTION controls how much of the bias direction is the latent
35# cluster signal vs per-particle noise. Higher α → more recoverable latent
36# structure → bigger CFL benefit in the ablation.
37#
38# Magnitude is modest (subdominant to the unit-length target direction) so
39# direction precision and loss remain meaningful: model still navigates toward
40# the target, bias just perturbs the heading. Recoverability of the latent
41# cluster signal comes from cluster members SHARING a target (handled in
42# game.py via per-true-cluster anchors), not from oversized bias.
43_BIAS_CLUSTER_FRACTION = 0.7
44_BIAS_MAG_LO = 0.25
45_BIAS_MAG_HI = 0.60
46
47# Identity slice used for KMeans assignment.
48# [0:2] direction is the persistent learned identity; [2] confidence is a slow EMA.
49# Features [3:8] are situational (pressure / alignment / loss / drift / stability)
50# and would cause spurious migrations if used for clustering.
51_IDENTITY_SLICE = slice(0, 2)
52
53# Hysteresis: a particle only migrates if the new cluster's loss is at least
54# this fraction lower than the current cluster's loss.
55# 0.0 = no hysteresis (every flip migrates), 0.5 = need 50% improvement,
56# 0.6 = need 60% (current — middle ground: moderate per-round churn while
57# letting IFCA recover purity in <30 rounds).
58_MIGRATION_HYSTERESIS = 0.6
59
60_BEHAVIORAL_FORCE = 30.0 # goal-directed push from model[0:2], scaled by confidence
61_STRAGGLER_LOSS = 0.60 # local_loss threshold to consider a particle stranded
62_STRAGGLER_DRIFT = 0.04 # drift_velocity threshold below which a particle is "stuck"
63
64_OBSTACLE_SOFT_ZONE = 20 # px beyond physical edge where pre-contact repulsion starts
65_OBSTACLE_SOFT_STRENGTH = 60.0 # base strength of the soft-zone push
66_BLEND_LOCAL_MATURE = 0.40 # local weight once matured (your existing value)
67
68# How strongly the emergent pull (net attract/repel force from neighbors)
69# biases the learned heading in local_train. The pull is normalized to a unit
70# vector first, then added to the ideal direction with this weight. ~1.5 means
71# emergent influence is comparable to the unit-length target direction —
72# enough to noticeably drag direction_precision when attraction physics is on.
73_EMERGENT_MODEL_WEIGHT = 1.5
