92 def _commit_edit(self, snap: SimSnapshot):
93 """@brief Commit the in-progress rule-cell edit to the simulation, if any.
94
95 @param snap Current snapshot (for context); the parsed value is sent via the command queue.
96 """
97 if self.editing_cell is None:
98 return
99 try:
100 # edit_buffer holds a config-scale value (e.g. 0.65); convert to the
101 # internal scale the simulator uses before sending it across.
102 cfg_val = max(-5.0, min(5.0, float(self.edit_buffer)))
103 scale = (snap.force_scale if snap else 1.0) or 1.0
104 i, j = self.editing_cell
105 self.cmd_queue.put({'type': 'set_rule', 'i': i, 'j': j, 'val': cfg_val * scale})
106 except (ValueError, TypeError):
107 pass
108 self.editing_cell = None
109 self.edit_buffer = ""

111 def _current_rule_val(self, canonical, snap: SimSnapshot) -> float:
112 """@brief Look up the current force for a canonical (i, j) pair, falling back to defaults.
113
114 @param canonical (min(i,j), max(i,j)) cluster-pair key.
115 @param snap Current snapshot holding the rules table and defaults.
116 @return float the effective force for that pair.
117 """
118 i, j = canonical
119 return snap.rules.get(canonical, snap.g_attract if i == j else snap.g_repel)

347 def run(self):
348 """@brief Main GUI loop: poll input, render the latest snapshot, and cap to FRAME_RATE."""
349 while self.game_running:
350 if self.sim._stop_event.is_set():
351 self.game_running = False
352 break
353
354 snap = self.sim.get_snapshot()
355
356 for event in pygame.event.get():
357 if event.type == pygame.QUIT:
358 self.game_running = False
359
360 elif event.type == pygame.MOUSEBUTTONDOWN and event.button == 1:
361 if self.bomb_rect.collidepoint(event.pos):
362 self._commit_edit(snap)
363 self.cmd_queue.put({'type': 'detonate'})
364 elif self.cfl_toggle_rect.collidepoint(event.pos):
365 self._commit_edit(snap)
366 self.cmd_queue.put({'type': 'toggle_cfl'})
367 elif self.attraction_toggle_rect.collidepoint(event.pos):
368 self._commit_edit(snap)
369 self.cmd_queue.put({'type': 'toggle_attraction'})
370 else:
371 now = pygame.time.get_ticks()
372 hit = None
373 for (i, j), rect in self.rule_cell_rects.items():
374 if rect.collidepoint(event.pos):
375 hit = (min(i, j), max(i, j))
376 break
377
378 if hit is not None:
379 if self.last_click_cell == hit and now - self.last_click_time < 400:
380 i, j = hit
381 self.cmd_queue.put({'type': 'reset_rule', 'i': i, 'j': j})
382 self.editing_cell = None
383 self.edit_buffer = ""
384 self.last_click_cell = None
385 else:
386 self._commit_edit(snap)
387 self.editing_cell = hit
388 self.edit_buffer = f"{self._current_rule_val(hit, snap) / (snap.force_scale or 1.0):.2f}"
389 self.last_click_cell = hit
390 self.last_click_time = now
391 else:
392 self._commit_edit(snap)
393 self.last_click_cell = None
394
395 elif event.type == pygame.KEYDOWN:
396 if event.key == pygame.K_t and self.editing_cell is None:
397 self.show_target_lines = not self.show_target_lines
398 elif self.editing_cell is not None:
399 if event.key in (pygame.K_RETURN, pygame.K_KP_ENTER):
400 self._commit_edit(snap)
401 elif event.key == pygame.K_ESCAPE:
402 self.editing_cell = None
403 self.edit_buffer = ""
404 elif event.key == pygame.K_BACKSPACE:
405 self.edit_buffer = self.edit_buffer[:-1]
406 elif event.unicode in "0123456789":
407 self.edit_buffer += event.unicode
408 elif event.unicode == "-" and self.edit_buffer == "":
409 self.edit_buffer = "-"
410 elif event.unicode == "." and "." not in self.edit_buffer:
411 self.edit_buffer += "."
412
413 if snap is not None:
414 self._draw_sim(snap)
415 self.draw_gui(snap)
416 pygame.display.flip()
417
418 self.clock.tick(FRAME_RATE)
