28class Particle:
29 """@brief A single federated agent: cognitive model + physical body.
30
31 Each particle is simultaneously a federated-learning *client* (it owns an
32 8-D @c model vector and trains it locally) and a physical body (position,
33 velocity) that moves under emergent forces. The two are coupled: the learned
34 heading drives motion, and motion/neighborhood feed back into the model.
35
36 Model layout (indices into @c self.model):
37 - [0,1] dir_x, dir_y — learned heading (unit vector); the clustering identity.
38 - [2] confidence — 0..1 EMA of how consistently it converges.
39 - [3] obstacle_pressure— 0..1 decaying memory of recent obstacle proximity.
40 - [4] peer_alignment — 0..1 cosine similarity to same-cluster neighbors.
41 - [5] rounds_stable — normalized rounds since the last cluster change.
42 - [6] local_loss — 0..1 distance + directional error to the target.
43 - [7] drift_velocity — 0..1 EMA of speed.
44 """
45
46 def __init__(self, x, v, c, target_idx, true_cluster_id: int,
47 cluster_bias_dir: Tuple[float, float], r=PARTICLE_DEFAULT_RADIUS):
48 """@brief Construct an agent with a random heading, target, and non-IID bias.
49
50 @param x (x, y) initial position.
51 @param v (vx, vy) initial velocity.
52 @param c RGB color tuple (overwritten by the cluster color at render time).
53 @param target_idx Index of the cluster target this agent initially chases.
54 @param true_cluster_id Ground-truth latent cluster label (the structure CFL is
55 meant to recover).
56 @param cluster_bias_dir Unit vector for this true cluster's shared bias direction.
57 @c _local_bias is drawn as a mix of this direction and per-particle
58 noise (see @c cfl_params.get_bias_cluster_fraction).
59 @param r Draw radius in pixels.
60 """
61 self.x = x[0]
62 self.y = x[1]
63 self.vx = v[0]
64 self.vy = v[1]
65 self.c = c
66 self.r = r
67 self.target_idx = target_idx
68 self.cluster_id = -1
69 self._true_cluster_id = true_cluster_id
70
71 direction = np.random.randn(2)
72 direction /= np.linalg.norm(direction)
73 self.model = np.array([
74 direction[0],
75 direction[1],
76 0.5, # start with middling confidence
77 0.0, # no obstacle pressure yet
78 0.0, # no peer alignment computed yet
79 0.0, # not stable yet
80 1.0, # assume far from target initially
81 0.0, # not moving yet
82 ], dtype=np.float64)
83
84 # Internal bookkeeping
85 self._prev_cluster_id = -1
86 self._stable_rounds = 0
87 self._speed_ema = 0.0 # exponential moving average of speed
88
89 # Per-particle inner target — what this particle actually chases.
90 # The group/cluster target is just the average of these for visualization.
91 _margin = WALL_BOUNDARY * 4
92 self._target_x = random.uniform(_margin, SIM_WIDTH - _margin)
93 self._target_y = random.uniform(_margin, SIM_HEIGHT - _margin)
94 self._target_angle = random.uniform(0, 2 * math.pi)
95
96 # Per-particle directional bias — the non-IID analog from the CFL paper.
97 # Each "client" systematically misperceives the ideal direction by a fixed
98 # offset. When a cluster_bias_dir is provided, the bias direction is a mix
99 # of that cluster-shared signal and per-particle noise — so members of the
100 # same true cluster share a latent direction that CFL aggregation can
101 # recover, while individual learning cannot. Magnitude stays random per
102 # particle so even within-cluster, no two agents are identical.
103 _bias_strength = random.uniform(cfl_params.get_bias_mag_lo(), cfl_params.get_bias_mag_hi())
104 _noise_angle = random.uniform(0, 2 * math.pi)
105 _nx, _ny = math.cos(_noise_angle), math.sin(_noise_angle)
106 _cx, _cy = cluster_bias_dir
107 _bias_frac = cfl_params.get_bias_cluster_fraction()
108 _mx = _bias_frac * _cx + (1.0 - _bias_frac) * _nx
109 _my = _bias_frac * _cy + (1.0 - _bias_frac) * _ny
110 _mn = math.hypot(_mx, _my)
111 if _mn > 0:
112 _mx, _my = _mx / _mn, _my / _mn
113 else:
114 _mx, _my = _nx, _ny
115 self._local_bias = np.array([_mx * _bias_strength, _my * _bias_strength])

118def local_train(particle, current_target_pos, obstacles, learning_rate=0.1,
119 emergent_pull=(0.0, 0.0)):
120 """@brief One local training step for a single agent (the cognitive update).
121
122 Nudges the agent's learned heading @c model[0:2] toward its ideal direction —
123 the true direction to its target corrupted by the agent's fixed non-IID bias,
124 then blended with obstacle-avoidance AND any net emergent pull (attract/
125 repel from neighbors) — and refreshes the situational model dimensions:
126 confidence [2], obstacle_pressure [3], local_loss [6] and drift_velocity
127 [7]. The loss uses the *true* (unbiased) target direction so that
128 federation correcting the bias is visible as a loss decrease.
129
130 Wiring the emergent pull into the ideal direction (not just into velocity,
131 where apply_physics_rules already does it) means the LEARNED MODEL also
132 reflects the physical push — direction_precision drops under
133 attraction_enabled=True, decoupling the cognitive and physical contributions
134 of the configuration.
135
136 @param particle The Particle to train; mutated in place.
137 @param current_target_pos (x, y) of this agent's personal inner target.
138 @param obstacles List of (ox, oy, radius) obstacle circles to avoid.
139 @param learning_rate Base step size for the heading update (scaled up under pressure).
140 @param emergent_pull (ex, ey) net emergent force vector for this particle this
141 step. Direction is what matters — the vector is normalized to a unit
142 direction before being added to the ideal with @c cfl_params.get_emergent_model_weight.
143 Pass (0.0, 0.0) when attraction_enabled=False to keep behavior
144 identical to the pre-emergent-model code.
145 @return None — @p particle.model is updated in place.
146 """
147 tx = current_target_pos[0] - particle.x
148 ty = current_target_pos[1] - particle.y
149 dist_t = math.hypot(tx, ty)
150
151 if dist_t > 0:
152 tx_n, ty_n = tx / dist_t, ty / dist_t
153 else:
154 tx_n, ty_n = 0.0, 0.0
155
156 # Save the unbiased target direction so we can compute true directional error later.
157 # (alignment vs. biased ideal would DECREASE when CFL corrects the model, which
158 # is the opposite of what we want the loss metric to show.)
159 _true_tx, _true_ty = tx_n, ty_n
160
161 # Non-IID bias: each particle persistently misperceives the ideal direction.
162 # Without CFL aggregation the bias accumulates; federation averages it out.
163 _bx = tx_n + particle._local_bias[0]
164 _by = ty_n + particle._local_bias[1]
165 _bn = math.hypot(_bx, _by)
166 if _bn > 0:
167 tx_n, ty_n = _bx / _bn, _by / _bn
168
169 # --- Obstacle repulsion ---
170 ox_total, oy_total = 0.0, 0.0
171 raw_pressure = 0.0
172 for ox, oy, orad in obstacles:
173 dx = particle.x - ox
174 dy = particle.y - oy
175 dist_o = math.hypot(dx, dy)
176 sense_radius = orad + 100 # wider sensing range so avoidance starts earlier
177 if 0 < dist_o < sense_radius:
178 t = (sense_radius - dist_o) / sense_radius # 0 at edge, 1 at surface
179 push = t * t * t # cubic: steep near surface, gentle at distance
180 ox_total += (dx / dist_o) * push
181 oy_total += (dy / dist_o) * push
182 raw_pressure += t # linear for EMA (represents zone depth, not push force)
183
184 # Obstacle weight scales up as particle gets closer — escape dominates target near surface
185 obstacle_weight = 1.25 + raw_pressure * 0.75
186 ideal_x = tx_n + ox_total * obstacle_weight
187 ideal_y = ty_n + oy_total * obstacle_weight
188
189 # Emergent pull: normalized neighbor-force direction added to the ideal so
190 # the learned heading also reflects where physics is dragging this agent.
191 # Magnitude of the input vector is discarded — only direction is kept, then
192 # scaled by cfl_params.get_emergent_model_weight(). Comparable in scale to the unit-length
193 # target direction so it noticeably bends model[0:2] when present.
194 ex_in, ey_in = emergent_pull
195 em_norm = math.hypot(ex_in, ey_in)
196 if em_norm > 0.0:
197 _emergent_w = cfl_params.get_emergent_model_weight()
198 ideal_x += (ex_in / em_norm) * _emergent_w
199 ideal_y += (ey_in / em_norm) * _emergent_w
200
201 norm = math.hypot(ideal_x, ideal_y)
202 if norm > 0:
203 ideal_x, ideal_y = ideal_x / norm, ideal_y / norm
204
205 # --- Update [0:2]: direction ---
206 # Learning rate scales with sustained pressure so the direction adapts faster when stuck
207 pressure_lr = min(learning_rate * (1.0 + particle.model[3] * 4.0), 0.5)
208 particle.model[0] += pressure_lr * (ideal_x - particle.model[0])
209 particle.model[1] += pressure_lr * (ideal_y - particle.model[1])
210 norm_m = np.linalg.norm(particle.model[0:2])
211 if norm_m > 0:
212 particle.model[0:2] /= norm_m
213
214 # --- Update [2]: confidence ---
215 alignment = particle.model[0] * ideal_x + particle.model[1] * ideal_y # -1..1
216 confidence_signal = (alignment + 1) / 2 # remap to 0..1
217 # Softer pressure penalty: killing confidence when stuck weakens the escape force
218 confidence_signal *= max(0.0, 1.0 - raw_pressure * 0.4)
219 particle.model[2] += 0.05 * (confidence_signal - particle.model[2])
220 particle.model[2] = np.clip(particle.model[2], 0.01, 1.0)
221
222 # --- Update [3]: obstacle_pressure (decaying memory) ---
223 particle.model[3] = 0.9 * particle.model[3] + 0.1 * min(raw_pressure, 1.0)
224
225 # --- Update [6]: local_loss (geometric distance + directional error) ---
226 # directional_error uses the TRUE target direction (pre-bias), not the biased ideal.
227 # When CFL corrects the model toward the true direction, true_alignment rises and
228 # this error falls — making the CFL benefit visible in the loss metric.
229 max_dist = math.hypot(SIM_DIM[0], SIM_DIM[1])
230 geo_loss = min(dist_t / max_dist, 1.0)
231 true_alignment = particle.model[0] * _true_tx + particle.model[1] * _true_ty
232 directional_error = 1.0 - (true_alignment + 1.0) / 2.0
233 particle.model[6] = min(0.5 * geo_loss + 0.5 * directional_error, 1.0)
234
235 # --- Update [7]: drift_velocity (EMA of speed) ---
236 speed = math.hypot(particle.vx, particle.vy)
237 particle._speed_ema = 0.95 * particle._speed_ema + 0.05 * speed
238 particle.model[7] = min(particle._speed_ema / PARTICLE_MAX_SPEED, 1.0)

240def update_peer_alignment(particles, neighbor_radius=90.0):
241 """@brief Feed spatial-neighborhood consensus back into each agent's model (physical -> cognitive).
242
243 Sets @c model[4] (peer_alignment) of every agent to the cosine similarity
244 between its own heading and the mean heading of same-cluster neighbors within
245 @p neighbor_radius, remapped to 0..1. Agents with no cluster or no neighbors
246 get 0.
247
248 @param particles List of all Particle agents; mutated in place.
249 @param neighbor_radius Pixel radius defining the spatial neighborhood.
250 @return None.
251 """
252 if not particles:
253 return
254
255 positions = np.array([(p.x, p.y) for p in particles])
256 directions = np.array([p.model[0:2] for p in particles])
257 cids = np.array([p.cluster_id for p in particles])
258
259 for i, p in enumerate(particles):
260 if p.cluster_id == -1:
261 p.model[4] = 0.0
262 continue
263
264 same_cluster = (cids == p.cluster_id)
265 same_cluster[i] = False
266
267 if not np.any(same_cluster):
268 p.model[4] = 0.0
269 continue
270
271 diffs = positions[same_cluster] - positions[i]
272 in_radius = np.linalg.norm(diffs, axis=1) < neighbor_radius
273
274 neighbor_dirs = directions[same_cluster][in_radius]
275 if len(neighbor_dirs) == 0:
276 p.model[4] = 0.0
277 continue
278
279 avg_dir = neighbor_dirs.mean(axis=0)
280 norm = np.linalg.norm(avg_dir)
281 if norm > 0:
282 avg_dir /= norm
283
284 cos_sim = float(np.dot(directions[i], avg_dir))
285 p.model[4] = (cos_sim + 1) / 2
