55# Non-IID bias structure: each particle's _local_bias direction is a mix of
56# its true cluster's shared bias direction and a per-particle random direction.
57# BIAS_CLUSTER_FRACTION controls how much of the bias direction is the latent
58# cluster signal vs per-particle noise. Higher α → more recoverable latent
59# structure → bigger CFL benefit in the ablation.
60#
61# Magnitude is modest (subdominant to the unit-length target direction) so
62# direction precision and loss remain meaningful: model still navigates toward
63# the target, bias just perturbs the heading. Recoverability of the latent
64# cluster signal comes from cluster members SHARING a target (handled in
65# game.py via per-true-cluster anchors), not from oversized bias.

98class Particle:
99 """@brief A single federated agent: cognitive model + physical body.
100
101 Each particle is simultaneously a federated-learning *client* (it owns an
102 8-D @c model vector and trains it locally) and a physical body (position,
103 velocity) that moves under emergent forces. The two are coupled: the learned
104 heading drives motion, and motion/neighbourhood feed back into the model.
105
106 Model layout (indices into @c self.model):
107 - [0,1] dir_x, dir_y — learned heading (unit vector); the clustering identity.
108 - [2] confidence — 0..1 EMA of how consistently it converges.
109 - [3] obstacle_pressure— 0..1 decaying memory of recent obstacle proximity.
110 - [4] peer_alignment — 0..1 cosine similarity to same-cluster neighbours.
111 - [5] rounds_stable — normalised rounds since the last cluster change.
112 - [6] local_loss — 0..1 distance + directional error to the target.
113 - [7] drift_velocity — 0..1 EMA of speed.
114 """

116 def __init__(self, x, v, c, target_idx, r=PARTICLE_DEFAULT_RADIUS,
117 true_cluster_id: int = -1, cluster_bias_dir: Tuple[float, float] = None):
118 """@brief Construct an agent with a random heading, target, and non-IID bias.
119
120 @param x (x, y) initial position.
121 @param v (vx, vy) initial velocity.
122 @param c RGB colour tuple (overwritten by the cluster colour at render time).
123 @param target_idx Index of the cluster target this agent initially chases.
124 @param r Draw radius in pixels.
125 @param true_cluster_id Ground-truth latent cluster label (the structure CFL is
126 meant to recover). -1 means "no ground truth" — bias falls back to fully
127 random and purity metrics will be N/A.
128 @param cluster_bias_dir Unit vector for this true cluster's shared bias direction.
129 When provided, @c _local_bias is drawn as a mix of this direction and per-particle
130 noise (see @c BIAS_CLUSTER_FRACTION). When None, bias is fully random (legacy).
131 """
132 self.x = x[0]
133 self.y = x[1]
134 self.vx = v[0]
135 self.vy = v[1]
136 self.c = c
137 self.r = r
138 self.target_idx = target_idx
139 self.cluster_id = -1
140 self._true_cluster_id = true_cluster_id
141
142 # 8-dimensional model:
143 # [0] dir_x, [1] dir_y — learned heading (unit vector, as before)
144 # [2] confidence — 0..1, how consistently this particle converges
145 # [3] obstacle_pressure — 0..1, decaying memory of recent obstacle hits
146 # [4] peer_alignment — 0..1, cosine sim to same-cluster neighbors
147 # [5] rounds_stable — normalised count of rounds without cluster change
148 # [6] local_loss — normalised distance to personal target
149 # [7] drift_velocity — normalised recent average speed
150 direction = np.random.randn(2)
151 direction /= np.linalg.norm(direction)
152 self.model = np.array([
153 direction[0],
154 direction[1],
155 0.5, # start with middling confidence
156 0.0, # no obstacle pressure yet
157 0.0, # no peer alignment computed yet
158 0.0, # not stable yet
159 1.0, # assume far from target initially
160 0.0, # not moving yet
161 ], dtype=np.float64)
162
163 # Internal bookkeeping
164 self._prev_cluster_id = -1
165 self._stable_rounds = 0
166 self._speed_ema = 0.0 # exponential moving average of speed
167
168 # Per-particle inner target — what this particle actually chases.
169 # The group/cluster target is just the average of these for visualisation.
170 _margin = WALL_BOUNDARY * 4
171 self._target_x = random.uniform(_margin, SIM_WIDTH - _margin)
172 self._target_y = random.uniform(_margin, SIM_HEIGHT - _margin)
173 self._target_angle = random.uniform(0, 2 * math.pi)
174
175 # Per-particle directional bias — the non-IID analog from the CFL paper.
176 # Each "client" systematically misperceives the ideal direction by a fixed
177 # offset. When a cluster_bias_dir is provided, the bias direction is a mix
178 # of that cluster-shared signal and per-particle noise — so members of the
179 # same true cluster share a latent direction that CFL aggregation can
180 # recover, while individual learning cannot. Magnitude stays random per
181 # particle so even within-cluster, no two agents are identical.
182 _bias_strength = random.uniform(BIAS_MAG_LO, BIAS_MAG_HI)
183 _noise_angle = random.uniform(0, 2 * math.pi)
184 _nx, _ny = math.cos(_noise_angle), math.sin(_noise_angle)
185 if cluster_bias_dir is not None:
186 _cx, _cy = cluster_bias_dir
187 _mx = BIAS_CLUSTER_FRACTION * _cx + (1.0 - BIAS_CLUSTER_FRACTION) * _nx
188 _my = BIAS_CLUSTER_FRACTION * _cy + (1.0 - BIAS_CLUSTER_FRACTION) * _ny
189 _mn = math.hypot(_mx, _my)
190 if _mn > 0:
191 _mx, _my = _mx / _mn, _my / _mn
192 else:
193 _mx, _my = _nx, _ny
194 else:
195 _mx, _my = _nx, _ny
196 self._local_bias = np.array([_mx * _bias_strength, _my * _bias_strength])

199def local_train(particle, current_target_pos, obstacles, learning_rate=0.1,
200 emergent_pull=(0.0, 0.0)):
201 """@brief One local training step for a single agent (the cognitive update).
202
203 Nudges the agent's learned heading @c model[0:2] toward its ideal direction —
204 the true direction to its target corrupted by the agent's fixed non-IID bias,
205 then blended with obstacle-avoidance AND any net emergent pull (attract/
206 repel from neighbours) — and refreshes the situational model dimensions:
207 confidence [2], obstacle_pressure [3], local_loss [6] and drift_velocity
208 [7]. The loss uses the *true* (unbiased) target direction so that
209 federation correcting the bias is visible as a loss decrease.
210
211 Wiring the emergent pull into the ideal direction (not just into velocity,
212 where apply_physics_rules already does it) means the LEARNED MODEL also
213 reflects the physical push — direction_precision drops under
214 attraction_enabled=True, decoupling the cognitive and physical contributions
215 of the ablation.
216
217 @param particle The Particle to train; mutated in place.
218 @param current_target_pos (x, y) of this agent's personal inner target.
219 @param obstacles List of (ox, oy, radius) obstacle circles to avoid.
220 @param learning_rate Base step size for the heading update (scaled up under pressure).
221 @param emergent_pull (ex, ey) net emergent force vector for this particle this
222 step. Direction is what matters — the vector is normalised to a unit
223 direction before being added to the ideal with @c EMERGENT_MODEL_WEIGHT.
224 Pass (0.0, 0.0) when attraction_enabled=False to keep behaviour
225 identical to the pre-emergent-model code.
226 @return None — @p particle.model is updated in place.
227 """
228 tx = current_target_pos[0] - particle.x
229 ty = current_target_pos[1] - particle.y
230 dist_t = math.hypot(tx, ty)
231
232 if dist_t > 0:
233 tx_n, ty_n = tx / dist_t, ty / dist_t
234 else:
235 tx_n, ty_n = 0.0, 0.0
236
237 # Save the unbiased target direction so we can compute true directional error later.
238 # (alignment vs. biased ideal would DECREASE when CFL corrects the model, which
239 # is the opposite of what we want the loss metric to show.)
240 _true_tx, _true_ty = tx_n, ty_n
241
242 # Non-IID bias: each particle persistently misperceives the ideal direction.
243 # Without CFL aggregation the bias accumulates; federation averages it out.
244 _bx = tx_n + particle._local_bias[0]
245 _by = ty_n + particle._local_bias[1]
246 _bn = math.hypot(_bx, _by)
247 if _bn > 0:
248 tx_n, ty_n = _bx / _bn, _by / _bn
249
250 # --- Obstacle repulsion ---
251 ox_total, oy_total = 0.0, 0.0
252 raw_pressure = 0.0
253 for ox, oy, orad in obstacles:
254 dx = particle.x - ox
255 dy = particle.y - oy
256 dist_o = math.hypot(dx, dy)
257 sense_radius = orad + 100 # wider sensing range so avoidance starts earlier
258 if 0 < dist_o < sense_radius:
259 t = (sense_radius - dist_o) / sense_radius # 0 at edge, 1 at surface
260 push = t * t * t # cubic: steep near surface, gentle at distance
261 ox_total += (dx / dist_o) * push
262 oy_total += (dy / dist_o) * push
263 raw_pressure += t # linear for EMA (represents zone depth, not push force)
264
265 # Obstacle weight scales up as particle gets closer — escape dominates target near surface
266 obstacle_weight = 1.25 + raw_pressure * 0.75
267 ideal_x = tx_n + ox_total * obstacle_weight
268 ideal_y = ty_n + oy_total * obstacle_weight
269
270 # Emergent pull: normalised neighbour-force direction added to the ideal so
271 # the learned heading also reflects where physics is dragging this agent.
272 # Magnitude of the input vector is discarded — only direction is kept, then
273 # scaled by EMERGENT_MODEL_WEIGHT. Comparable in scale to the unit-length
274 # target direction so it noticeably bends model[0:2] when present.
275 ex_in, ey_in = emergent_pull
276 em_norm = math.hypot(ex_in, ey_in)
277 if em_norm > 0.0:
278 ideal_x += (ex_in / em_norm) * EMERGENT_MODEL_WEIGHT
279 ideal_y += (ey_in / em_norm) * EMERGENT_MODEL_WEIGHT
280
281 norm = math.hypot(ideal_x, ideal_y)
282 if norm > 0:
283 ideal_x, ideal_y = ideal_x / norm, ideal_y / norm
284
285 # --- Update [0:2]: direction ---
286 # Learning rate scales with sustained pressure so the direction adapts faster when stuck
287 pressure_lr = min(learning_rate * (1.0 + particle.model[3] * 4.0), 0.5)
288 particle.model[0] += pressure_lr * (ideal_x - particle.model[0])
289 particle.model[1] += pressure_lr * (ideal_y - particle.model[1])
290 norm_m = np.linalg.norm(particle.model[0:2])
291 if norm_m > 0:
292 particle.model[0:2] /= norm_m
293
294 # --- Update [2]: confidence ---
295 alignment = particle.model[0] * ideal_x + particle.model[1] * ideal_y # -1..1
296 confidence_signal = (alignment + 1) / 2 # remap to 0..1
297 # Softer pressure penalty: killing confidence when stuck weakens the escape force
298 confidence_signal *= max(0.0, 1.0 - raw_pressure * 0.4)
299 particle.model[2] += 0.05 * (confidence_signal - particle.model[2])
300 particle.model[2] = np.clip(particle.model[2], 0.01, 1.0)
301
302 # --- Update [3]: obstacle_pressure (decaying memory) ---
303 particle.model[3] = 0.9 * particle.model[3] + 0.1 * min(raw_pressure, 1.0)
304
305 # --- Update [6]: local_loss (geometric distance + directional error) ---
306 # directional_error uses the TRUE target direction (pre-bias), not the biased ideal.
307 # When CFL corrects the model toward the true direction, true_alignment rises and
308 # this error falls — making the CFL benefit visible in the loss metric.
309 max_dist = math.hypot(SIM_DIM[0], SIM_DIM[1])
310 geo_loss = min(dist_t / max_dist, 1.0)
311 true_alignment = particle.model[0] * _true_tx + particle.model[1] * _true_ty
312 directional_error = 1.0 - (true_alignment + 1.0) / 2.0
313 particle.model[6] = min(0.5 * geo_loss + 0.5 * directional_error, 1.0)
314
315 # --- Update [7]: drift_velocity (EMA of speed) ---
316 speed = math.hypot(particle.vx, particle.vy)
317 particle._speed_ema = 0.95 * particle._speed_ema + 0.05 * speed
318 particle.model[7] = min(particle._speed_ema / PARTICLE_MAX_SPEED, 1.0)

320def update_peer_alignment(particles, neighbor_radius=90.0):
321 """@brief Feed spatial-neighbourhood consensus back into each agent's model (physical -> cognitive).
322
323 Sets @c model[4] (peer_alignment) of every agent to the cosine similarity
324 between its own heading and the mean heading of same-cluster neighbours within
325 @p neighbor_radius, remapped to 0..1. Agents with no cluster or no neighbours
326 get 0.
327
328 @param particles List of all Particle agents; mutated in place.
329 @param neighbor_radius Pixel radius defining the spatial neighbourhood.
330 @return None.
331 """
332 if not particles:
333 return
334
335 n = len(particles)
336 positions = np.array([(p.x, p.y) for p in particles]) # (n, 2)
337 directions = np.array([p.model[0:2] for p in particles]) # (n, 2)
338 cids = np.array([p.cluster_id for p in particles]) # (n,)
339
340 for i, p in enumerate(particles):
341 if p.cluster_id == -1:
342 p.model[4] = 0.0
343 continue
344
345 same_cluster = (cids == p.cluster_id)
346 same_cluster[i] = False
347
348 if not np.any(same_cluster):
349 p.model[4] = 0.0
350 continue
351
352 diffs = positions[same_cluster] - positions[i] # (k, 2)
353 in_radius = np.linalg.norm(diffs, axis=1) < neighbor_radius
354
355 neighbor_dirs = directions[same_cluster][in_radius] # (m, 2)
356 if len(neighbor_dirs) == 0:
357 p.model[4] = 0.0
358 continue
359
360 avg_dir = neighbor_dirs.mean(axis=0)
361 norm = np.linalg.norm(avg_dir)
362 if norm > 0:
363 avg_dir /= norm
364
365 cos_sim = float(np.dot(directions[i], avg_dir))
366 p.model[4] = (cos_sim + 1) / 2

368def apply_physics_rules(particles: List[Particle], obstacles: List[Tuple[int, int, int]], g_attract: float, g_repel: float, dt: float, rules: dict = None, attraction_enabled: bool = True):
369 """@brief Integrate one physics tick: inter-agent forces, obstacles, walls (the physical domain).
370
371 Accumulates per-agent forces from: pairwise attraction/repulsion (governed by
372 cluster membership and the @p rules matrix), always-on emergency anti-stacking
373 repulsion, soft obstacle pre-contact repulsion, and the model-directed
374 "behavioral force" (cognitive -> physical coupling, scaled by confidence). It
375 then advances velocities/positions and resolves hard obstacle and wall collisions.
376
377 @param particles List of all Particle agents; positions/velocities mutated in place.
378 @param obstacles List of (ox, oy, radius) obstacle circles.
379 @param g_attract Default intra-cluster force (negative attracts).
380 @param g_repel Default inter-cluster force (positive repels).
381 @param dt Integration timestep in seconds.
382 @param rules Optional dict mapping (min(ci,cj), max(ci,cj)) -> float for per-pair forces.
383 @param attraction_enabled If False, intra/inter-cluster forces are disabled and only the
384 emergency anti-stacking repulsion remains — used to ablate the emergent physics.
385 @return None.
386 """
387 # Initialize forces
388 forces = [np.zeros(2) for _ in particles]
389
390 # Calculate interactions
391 for i in range(len(particles)):
392 a = particles[i]
393 for j in range(i + 1, len(particles)):
394 b = particles[j]
395
396 dx = a.x - b.x
397 dy = a.y - b.y
398 d_sq = dx ** 2 + dy ** 2
399
400 # 1. VISUAL OPTIMIZATION: If they are too far, skip math entirely
401 if d_sq > PARTICLE_FORCE_UPPER_RANGE ** 2:
402 continue
403
404 # Avoid division by zero
405 if d_sq == 0:
406 d_sq = 0.001
407
408 d = d_sq ** 0.5
409
410 # 2. EMERGENCY REPULSION (always on — prevents visual stacking)
411 if d < PARTICLE_FORCE_LOWER_RANGE * 2:
412 F_scalar = 200.0 # Massive repulsion to unclump them
413
414 # 3. NORMAL ATTRACTION/REPULSION (only when emergent rules are enabled)
415 elif attraction_enabled:
416 g = 0.0
417
418 # Only interact if both have a valid cluster
419 if a.cluster_id != -1 and b.cluster_id != -1:
420 ci, cj = a.cluster_id, b.cluster_id
421 if rules is not None:
422 key = (min(ci, cj), max(ci, cj))
423 default = g_attract if ci == cj else g_repel
424 g = rules.get(key, default)
425 elif ci == cj:
426 g = g_attract
427 else:
428 g = g_repel
429
430 # Standard Gravity Formula
431 denom = (d ** PARTICLE_POWER_OF_DISTANCE) * len(particles)
432 if denom == 0: denom = 0.001
433 F_scalar = g * (1 / denom)
434 else:
435 continue # attraction off and outside emergency range — no force
436
437 # Apply forces
438 fx = F_scalar * dx
439 fy = F_scalar * dy
440
441 forces[i] += np.array([fx, fy])
442 forces[j] -= np.array([fx, fy])
443
444 # Soft obstacle zone: pre-contact repulsion so particles redirect before touching.
445 # Quadratic falloff — very strong at the physical edge, tapers over OBSTACLE_SOFT_ZONE px.
446 for i, p in enumerate(particles):
447 for ox, oy, orad in obstacles:
448 dx, dy = p.x - ox, p.y - oy
449 dist = math.hypot(dx, dy)
450 if dist == 0:
451 continue
452 edge = orad + p.r
453 if dist < edge + OBSTACLE_SOFT_ZONE:
454 t = max(0.0, (edge + OBSTACLE_SOFT_ZONE - dist) / OBSTACLE_SOFT_ZONE)
455 push = OBSTACLE_SOFT_STRENGTH * t * t
456 forces[i][0] += (dx / dist) * push
457 forces[i][1] += (dy / dist) * push
458
459 # Behavioral force: model-directed push scaled by confidence.
460 # model[3] (obstacle_pressure) amplifies the force so sustained pressure
461 # translates directly into a stronger escape push.
462 for i, p in enumerate(particles):
463 if p.cluster_id != -1 and p.model[2] > 0.15:
464 pressure_boost = 1.0 + float(p.model[3]) * 1.25
465 strength = BEHAVIORAL_FORCE * float(p.model[2]) * pressure_boost
466 forces[i][0] += p.model[0] * strength
467 forces[i][1] += p.model[1] * strength
468
469 # Apply forces to velocities and positions
470 for i, p in enumerate(particles):
471 fx, fy = forces[i]
472
473 p.vx = (p.vx + fx * dt) * PARTICLE_LOSE_ENERGY
474 p.vy = (p.vy + fy * dt) * PARTICLE_LOSE_ENERGY
475
476 speed = (p.vx ** 2 + p.vy ** 2) ** 0.5
477
478 # If going faster than max, scale it down
479 if speed > PARTICLE_MAX_SPEED:
480 scale = PARTICLE_MAX_SPEED / speed
481 p.vx *= scale
482 p.vy *= scale
483
484 p.x += p.vx * dt
485 p.y += p.vy * dt
486
487 # --- HARD PHYSICAL COLLISIONS WITH OBSTACLES ---
488 for ox, oy, orad in obstacles:
489 dx, dy = p.x - ox, p.y - oy
490 dist = math.hypot(dx, dy)
491
492 # If they physically hit the obstacle, push them out
493 if dist < orad + p.r:
494 overlap = (orad + p.r) - dist
495 if dist > 0:
496 nx, ny = dx / dist, dy / dist
497 p.x += nx * overlap
498 p.y += ny * overlap
499 # Reflect velocity along surface normal (only if moving toward obstacle)
500 dot = p.vx * nx + p.vy * ny
501 if dot < 0:
502 p.vx -= 2 * dot * nx
503 p.vy -= 2 * dot * ny
504 p.vx *= 0.8
505 p.vy *= 0.8
506
507 # --- UPDATED WALL COLLISIONS ---
508 # Use SIM_DIM[0] and SIM_DIM[1] instead of SCREEN_DIM
509 V = 0.9
510 D = WALL_BOUNDARY
511
512 # Left Wall
513 if p.x < D:
514 p.x = D
515 p.vx *= -V
516
517 # Right Wall (Now uses simulation width, not window width)
518 if p.x > SIM_DIM[0] - D:
519 p.x = SIM_DIM[0] - D
520 p.vx *= -V
521
522 # Top Wall
523 if p.y < D:
524 p.y = D
525 p.vy *= -V
526
527 # Bottom Wall
528 if p.y > SIM_DIM[1] - D:
529 p.y = SIM_DIM[1] - D
530 p.vy *= -V

532def compute_cluster_stats(particles, n_clusters):
533 """@brief Aggregate per-cluster health used by the split/merge logic and logger.
534
535 @param particles List of all Particle agents.
536 @param n_clusters Number of clusters to report on (ids 0..n_clusters-1).
537 @return dict cluster_id -> {size, avg_loss, avg_confidence, mean_model}, where
538 @c mean_model is the element-wise mean 8-D model (None for empty clusters).
539 """
540 stats = {}
541 for cid in range(n_clusters):
542 members = [p for p in particles if p.cluster_id == cid]
543 if not members:
544 stats[cid] = {'size': 0, 'avg_loss': 0.0,
545 'avg_confidence': 0.0, 'mean_model': None}
546 continue
547 avg_loss = sum(p.model[6] for p in members) / len(members)
548 avg_conf = sum(p.model[2] for p in members) / len(members)
549 mean_model = sum(p.model for p in members) / len(members)
550 stats[cid] = {
551 'size': len(members),
552 'avg_loss': avg_loss,
553 'avg_confidence': avg_conf,
554 'mean_model': mean_model,
555 }
556 return stats

558def _direction_coherence(members):
559 """@brief Directional coherence of a cluster — the IFCA split signal.
560
561 Mean cosine similarity of members' headings to the cluster's own mean
562 heading: 1.0 = perfectly aligned, 0 = chaotic, -1 = anti-aligned. Under IFCA
563 avg_loss is pinned near the local minimum after assignment, so loss cannot
564 reveal an incoherent cluster — directional disagreement among members can.
565
566 @param members List of Particle agents in one cluster.
567 @return float coherence in [-1, 1] (1.0 when fewer than two members).
568 """
569 if len(members) < 2:
570 return 1.0
571 dirs = np.array([p.model[0:2] for p in members])
572 mean_dir = dirs.mean(axis=0)
573 norm = np.linalg.norm(mean_dir)
574 if norm == 0:
575 return 0.0
576 mean_dir = mean_dir / norm
577 return float((dirs @ mean_dir).mean())

580def _ifca_score(particle, target, theta, max_dist):
581 """@brief IFCA loss for evaluating cluster k on a single agent (lower is better).
582
583 Mirrors the per-agent loss from local_train() (model[6]) but evaluated
584 under cluster k's *broadcast* model theta and cluster k's target. This is the
585 score each IFCA client computes locally for every candidate model in order to
586 pick its cluster (argmin over k).
587
588 @param particle The agent being evaluated.
589 @param target (x, y) spatial target of cluster k.
590 @param theta Cluster k's broadcast 8-D model (its heading is theta[0:2]); may be None.
591 @param max_dist Diagonal of the simulation area, used to normalise distance.
592 @return float loss = 0.5 * geometric_loss + 0.5 * directional_loss.
593 """
594 dx = target[0] - particle.x
595 dy = target[1] - particle.y
596 dist = math.hypot(dx, dy)
597 geo_loss = min(dist / max_dist, 1.0)
598
599 if dist > 0 and theta is not None:
600 true_dx, true_dy = dx / dist, dy / dist
601 # Alignment of θ_k with the true direction from particle to target_k.
602 alignment = float(theta[0] * true_dx + theta[1] * true_dy) # -1..1
603 dir_loss = (1.0 - alignment) / 2.0
604 else:
605 dir_loss = 0.5 # no info → neutral
606
607 return 0.5 * geo_loss + 0.5 * dir_loss

610def _compute_cluster_models(particles, n_clusters):
611 """@brief Confidence-weighted mean model per cluster — the broadcast theta_k.
612
613 Each cluster's model is the mean of its members' 8-D models weighted by
614 confidence (model[2]), with the heading [0:2] re-normalised to a unit vector.
615
616 @param particles List of all Particle agents.
617 @param n_clusters Number of clusters.
618 @return list of length @p n_clusters; each entry is a length-8 numpy array, or
619 None for an empty cluster.
620 """
621 cluster_models = [None] * n_clusters
622 for k in range(n_clusters):
623 members = [p for p in particles if p.cluster_id == k]
624 if not members:
625 continue
626 weights = np.array([p.model[2] for p in members])
627 weights /= weights.sum() + 1e-8
628 theta = sum(w * p.model for w, p in zip(weights, members))
629 norm = np.linalg.norm(theta[0:2])
630 if norm > 0:
631 theta[0:2] = theta[0:2] / norm
632 cluster_models[k] = theta
633 return cluster_models

636def run_cfl_round(particles, kmeans_model, cluster_targets, cluster_colors, cluster_ages, cooldown_counter):
637 """@brief One IFCA-style clustered federated learning round (Ghosh et al. 2020).
638
639 Steps: (1) broadcast a confidence-weighted model theta_k per cluster; (2) each
640 agent re-assigns to the cluster whose theta_k gives the lowest loss (argmin),
641 subject to migration hysteresis; (3) aggregate members toward theta_k with an
642 age-adaptive blend; (4) rescue stuck stragglers; (5) after a cooldown, possibly
643 @b merge redundant/tiny clusters or @b split an incoherent one. A fresh KMeans
644 is fit at the end purely to preserve the @c .n_clusters / @c .inertia_ contract
645 that the logger reads — IFCA itself does not need it.
646
647 @param particles List of all Particle agents; cluster_id and model mutated in place.
648 @param kmeans_model Previous KMeans (read for @c .n_clusters; re-fit and returned).
649 @param cluster_targets List of (x, y) spatial targets, one per cluster.
650 @param cluster_colors dict cluster_id -> RGB (key -1 reserved for unassigned).
651 @param cluster_ages dict cluster_id -> age in rounds (drives the blend ratio).
652 @param cooldown_counter Rounds remaining before another split/merge may fire.
653 @return 8-tuple (transfers, kmeans_model, cluster_targets, cluster_colors,
654 cluster_ages, num_clusters, event, cooldown_counter), where @c transfers maps
655 (src, dst) -> migration count and @c event is None, 'split', or ('merge', dropped_id).
656 """
657 if not particles:
658 return {}, kmeans_model, cluster_targets, cluster_colors, cluster_ages, kmeans_model.n_clusters, None, cooldown_counter
659
660 n = kmeans_model.n_clusters
661 old_ids = [p.cluster_id for p in particles]
662
663 # --- IFCA broadcast: build θ_k from current cluster membership ---
664 cluster_models = _compute_cluster_models(particles, n)
665 max_dist = math.hypot(SIM_DIM[0], SIM_DIM[1])
666
667 # Fallback for clusters that are currently empty: seed θ_k with a
668 # unit vector pointing from the simulation centre toward target_k,
669 # so the loss is well-defined and the cluster can still attract members.
670 cx, cy = SIM_DIM[0] / 2.0, SIM_DIM[1] / 2.0
671 for k in range(n):
672 if cluster_models[k] is None and k < len(cluster_targets):
673 tx, ty = cluster_targets[k]
674 dx, dy = tx - cx, ty - cy
675 d = math.hypot(dx, dy)
676 theta = np.zeros(8, dtype=np.float64)
677 if d > 0:
678 theta[0], theta[1] = dx / d, dy / d
679 theta[2] = 0.5 # middling confidence
680 cluster_models[k] = theta
681
682 # --- IFCA assignment: each particle picks argmin loss across θ_k ---
683 transfers = defaultdict(int)
684 for i, p in enumerate(particles):
685 old_id = old_ids[i]
686
687 losses = [
688 _ifca_score(p, cluster_targets[k], cluster_models[k], max_dist)
689 for k in range(n)
690 ]
691 new_id = int(np.argmin(losses))
692
693 # Hysteresis: only switch if the chosen cluster's loss is meaningfully
694 # below the current cluster's loss. Stops particles flipping between
695 # nearly-equivalent clusters when geometry barely favours one.
696 if 0 <= old_id < n and old_id != new_id:
697 if losses[new_id] >= losses[old_id] * (1.0 - MIGRATION_HYSTERESIS):
698 new_id = old_id
699
700 if old_id != new_id:
701 transfers[(old_id, new_id)] += 1
702 p.cluster_id = new_id
703
704 # --- Aggregation: recompute θ_k from new assignments and blend ---
705 aggregated_models = _compute_cluster_models(particles, n)
706 for cid in range(n):
707 aggregated = aggregated_models[cid]
708 if aggregated is None:
709 continue
710 members = [p for p in particles if p.cluster_id == cid]
711
712 # Blend ratio depends on cluster age (newborn clusters keep more
713 # local state so they don't immediately re-merge with the parent).
714 age = cluster_ages.get(cid, BLEND_MATURITY_ROUNDS)
715 t = min(age / BLEND_MATURITY_ROUNDS, 1.0) # 0.0 = newborn, 1.0 = mature
716 local_weight = BLEND_LOCAL_NEW + t * (BLEND_LOCAL_MATURE - BLEND_LOCAL_NEW)
717 global_weight = 1.0 - local_weight
718
719 for p in members:
720 p.model = local_weight * p.model + global_weight * aggregated
721 p.model[2] = np.clip(p.model[2], 0.01, 1.0)
722 p.model[3:5] = np.clip(p.model[3:5], 0.0, 1.0)
723
724 # --- Re-fit KMeans purely for the .inertia_ / .n_clusters contract ---
725 # IFCA itself doesn't need this; downstream logging/printing reads it.
726 try:
727 identity_models = np.array([p.model[IDENTITY_SLICE] for p in particles])
728 init_centroids = np.array([
729 np.mean([p.model[IDENTITY_SLICE] for p in particles if p.cluster_id == cid], axis=0)
730 if any(p.cluster_id == cid for p in particles)
731 else identity_models[np.random.randint(len(identity_models))]
732 for cid in range(n)
733 ])
734 warm_kmeans = KMeans(n_clusters=n, init=init_centroids, n_init=1, random_state=0)
735 warm_kmeans.fit(identity_models)
736 kmeans_model = warm_kmeans
737 except Exception:
738 pass
739
740 # Stable round counter
741 for p in particles:
742 if p.cluster_id == p._prev_cluster_id:
743 p._stable_rounds = min(p._stable_rounds + 1, 50)
744 else:
745 p._stable_rounds = 0
746 p._prev_cluster_id = p.cluster_id
747 p.model[5] = p._stable_rounds / 50.0
748
749 # Increment age for all existing clusters
750 for cid in range(n):
751 cluster_ages[cid] = cluster_ages.get(cid, 0) + 1
752
753 # Straggler rescue: particles with high loss and near-zero drift are stuck.
754 # Re-assign them to the spatially nearest cluster target every round
755 # (runs even during cooldown so stuck particles aren't left orphaned).
756 for p in particles:
757 if p.model[6] > STRAGGLER_LOSS and p.model[7] < STRAGGLER_DRIFT:
758 nearest = min(
759 range(len(cluster_targets)),
760 key=lambda i: math.hypot(p.x - cluster_targets[i][0],
761 p.y - cluster_targets[i][1])
762 )
763 if nearest != p.cluster_id:
764 p.cluster_id = nearest
765 p.target_idx = nearest
766 p._prev_cluster_id = -1
767 p._stable_rounds = 0
768
769 # --- Restructuring ---
770 if cooldown_counter > 0:
771 return transfers, kmeans_model, cluster_targets, cluster_colors, cluster_ages, n, None, cooldown_counter - 1
772
773 stats = compute_cluster_stats(particles, n)
774
775 # MERGE
776 if n > MIN_CLUSTERS:
777 merge_pair = None
778
779 # Path 1: Absorb tiny clusters. A cluster with fewer than MIN_CLUSTER_SIZE
780 # members is effectively dead under IFCA (its θ_k is too noisy) — fold
781 # it into the cluster whose target is nearest, regardless of age.
782 tiny = [cid for cid in range(n) if stats[cid]['size'] < MIN_CLUSTER_SIZE]
783 for tcid in tiny:
784 tx, ty = cluster_targets[tcid]
785 other = [cid for cid in range(n) if cid != tcid and stats[cid]['size'] > 0]
786 if not other:
787 continue
788 nearest = min(
789 other,
790 key=lambda cid: math.hypot(cluster_targets[cid][0] - tx,
791 cluster_targets[cid][1] - ty),
792 )
793 merge_pair = (min(nearest, tcid), max(nearest, tcid))
794 print(f" > MERGE-tiny: cluster {tcid} (size={stats[tcid]['size']}) absorbed into {nearest}")
795 break
796
797 # Path 2: Standard similarity-based merge for mature, redundant clusters.
798 if merge_pair is None:
799 best_score, candidate = -float('inf'), None
800 for a in range(n):
801 for b in range(a + 1, n):
802 ma = stats[a]['mean_model']
803 mb = stats[b]['mean_model']
804 if ma is None or mb is None:
805 continue
806 if cluster_ages.get(a, 0) < BLEND_MATURITY_ROUNDS or \
807 cluster_ages.get(b, 0) < BLEND_MATURITY_ROUNDS:
808 continue
809
810 sim = float(np.dot(ma[0:2], mb[0:2]))
811 # Spatial-target component: 1.0 when targets coincide,
812 # 0.0 when they're MERGE_TARGET_DISTANCE or farther apart.
813 tdist = math.hypot(
814 cluster_targets[a][0] - cluster_targets[b][0],
815 cluster_targets[a][1] - cluster_targets[b][1],
816 )
817 target_score = max(0.0, 1.0 - tdist / MERGE_TARGET_DISTANCE)
818 # Combined score: high direction sim OR very close targets.
819 score = max(sim, target_score)
820 if score > best_score:
821 best_score, candidate = score, (a, b)
822
823 if candidate and best_score >= MERGE_SIMILARITY_THRESHOLD:
824 merge_pair = candidate
825 print(f" > MERGE-redundant: {candidate} score={best_score:.3f}")
826
827 if merge_pair:
828 keep, drop = merge_pair
829 for p in particles:
830 if p.cluster_id == drop:
831 p.cluster_id = keep
832 p._prev_cluster_id = -1 # force stability reset next round
833 p._stable_rounds = 0
834 for p in particles:
835 if p.cluster_id > drop:
836 p.cluster_id -= 1
837 p._prev_cluster_id = p.cluster_id
838
839 new_targets = [t for i, t in enumerate(cluster_targets) if i != drop]
840 new_colors = {(i if i < drop else i - 1): c
841 for i, c in cluster_colors.items() if i != drop and i != -1}
842 new_colors[-1] = cluster_colors[-1]
843
844 # Remap ages, drop the merged cluster
845 new_ages = {}
846 for cid, age in cluster_ages.items():
847 if cid == drop:
848 continue
849 new_cid = cid if cid < drop else cid - 1
850 new_ages[new_cid] = age
851 # The surviving cluster inherits the older age
852 new_ages[keep if keep < drop else keep - 1] = max(
853 cluster_ages.get(keep, 0), cluster_ages.get(drop, 0)
854 )
855
856 new_n = n - 1
857 new_kmeans = KMeans(n_clusters=new_n, n_init=10, random_state=0)
858
859 max_target_idx = len(new_targets) - 1
860 for p in particles:
861 p.target_idx = min(p.target_idx, max_target_idx)
862
863 return transfers, new_kmeans, new_targets, new_colors, new_ages, new_n, ('merge', drop), 10
864
865 # SPLIT — under IFCA the right signal is directional incoherence among
866 # members (avg_loss is mechanically pinned at the local min after assign).
867 if n < MAX_CLUSTERS:
868 eligible = [cid for cid in range(n)
869 if stats[cid]['size'] >= MIN_CLUSTER_SIZE * 2]
870
871 worst_cid, worst_score = None, 0.0
872 for cid in eligible:
873 members = [p for p in particles if p.cluster_id == cid]
874 coherence = _direction_coherence(members)
875 avg_loss = stats[cid]['avg_loss']
876
877 # Combined "wants to split" score: how much each criterion exceeds
878 # its threshold. Either alone can trigger.
879 coh_excess = max(0.0, SPLIT_COHERENCE_THRESHOLD - coherence)
880 loss_excess = max(0.0, avg_loss - SPLIT_LOSS_THRESHOLD)
881 score = coh_excess + loss_excess
882 if score > worst_score:
883 worst_score, worst_cid = score, cid
884
885 if worst_cid is not None and worst_score > 0.0:
886 members = [p for p in particles if p.cluster_id == worst_cid]
887 members_coh = _direction_coherence(members)
888 members_avg = stats[worst_cid]['avg_loss']
889 print(f" > SPLIT-trigger: cluster {worst_cid} "
890 f"coherence={members_coh:.2f} avg_loss={members_avg:.2f}")
891
892 # Split along the direction-disagreement axis: 2-means on member
893 # heading vectors. Members in one heading group form the new cluster.
894 # Falls back to spatial-x split when 2-means is too unbalanced.
895 new_members = None
896 if len(members) >= MIN_CLUSTER_SIZE * 2:
897 try:
898 dirs = np.array([p.model[0:2] for p in members])
899 km2 = KMeans(n_clusters=2, n_init=3, random_state=0)
900 labels = km2.fit_predict(dirs)
901 group_a = [m for m, l in zip(members, labels) if l == 0]
902 group_b = [m for m, l in zip(members, labels) if l == 1]
903 if len(group_a) >= MIN_CLUSTER_SIZE and len(group_b) >= MIN_CLUSTER_SIZE:
904 # Smaller group becomes the new cluster.
905 new_members = group_b if len(group_b) <= len(group_a) else group_a
906 except Exception:
907 pass
908
909 if new_members is None:
910 members_sorted = sorted(members, key=lambda p: p.x)
911 split_count = max(MIN_CLUSTER_SIZE, len(members_sorted) // 2)
912 new_members = members_sorted[:split_count]
913
914 new_cid = n
915 for p in new_members:
916 p.cluster_id = new_cid
917 p._prev_cluster_id = -1
918 p._stable_rounds = 0
919
920 ox, oy = cluster_targets[worst_cid]
921 new_targets = cluster_targets + [(
922 int(np.clip(ox + random.randint(-150, 150), 50, SIM_WIDTH - 50)),
923 int(np.clip(oy + random.randint(-150, 150), 50, SIM_HEIGHT - 50))
924 )]
925 used = set(cluster_colors.values()) - {cluster_colors[-1]}
926 new_colors = dict(cluster_colors)
927 new_colors[new_cid] = _pick_unused_color(used)
928
929 # New cluster starts at age 0, parent keeps its age
930 new_ages = dict(cluster_ages)
931 new_ages[new_cid] = 0
932
933 new_n = n + 1
934 new_kmeans = KMeans(n_clusters=new_n, n_init=10, random_state=0)
935
936 new_cid = n
937 for p in particles:
938 if p.cluster_id == new_cid:
939 p.target_idx = len(new_targets) - 1
940
941 return transfers, new_kmeans, new_targets, new_colors, new_ages, new_n, 'split', 10
942
943 return transfers, kmeans_model, cluster_targets, cluster_colors, cluster_ages, n, None, 0
