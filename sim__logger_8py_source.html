69def _cluster_purity(particles) -> Optional[float]:
70 """@brief Clustering purity of predicted @c cluster_id against ground-truth @c _true_cluster_id.
71
72 Purity = (1/N) * Σ_k max_j |{p : p.cluster_id = k} ∩ {p : p._true_cluster_id = j}|.
73
74 Standard unsupervised-clustering metric: for each predicted cluster, count
75 how many of its members share the dominant ground-truth label, sum across
76 clusters, divide by N. Headline ablation metric here — CFL aggregation
77 drives it toward 1.0 by discovering the latent bias structure, while
78 individual learning leaves @c cluster_id pinned at the random init so purity
79 hovers near 1/K.
80
81 Returns None when no particle carries a ground-truth label (legacy runs);
82 callers should treat that as N/A rather than logging a fake 1.0.
83
84 @param particles List of Particle agents.
85 @return float in [0, 1] or None.
86 """
87 if not particles:
88 return None
89 has_truth = any(getattr(p, "_true_cluster_id", -1) >= 0 for p in particles)
90 if not has_truth:
91 return None
92 counts: Dict[int, Dict[int, int]] = defaultdict(lambda: defaultdict(int))
93 for p in particles:
94 true_cid = getattr(p, "_true_cluster_id", -1)
95 if true_cid < 0:
96 continue
97 counts[p.cluster_id][true_cid] += 1
98 matched = sum(max(true_counts.values()) for true_counts in counts.values())
99 return matched / len(particles)

102def _spatial_precision(particle, max_dist: float) -> float:
103 """@brief How close a particle is to its target, in [0, 1] (1 = at the target).
104
105 @c spatial_precision = 1 - dist(particle, _target) / sim_diagonal. Same scale
106 as @c direction_precision so the two metrics are visually comparable on the
107 same axes. Complements @c direction_precision: the model can be aimed
108 correctly (high dir_prec) while the body is dragged off-target by emergent
109 forces (low spatial_prec) — the two precisions decouple the cognitive and
110 physical errors.
111
112 @param particle A Particle with @c x, @c y, @c _target_x, @c _target_y.
113 @param max_dist Sim diagonal (normaliser; same one used by local_loss).
114 @return float in [0, 1].
115 """
116 dx = particle._target_x - particle.x
117 dy = particle._target_y - particle.y
118 dist = math.hypot(dx, dy)
119 return float(1.0 - min(dist / max_dist, 1.0))

122def _direction_precision(particle) -> float:
123 """@brief Cosine alignment of a particle's learned heading to the *true* (unbiased) target direction, mapped to [0, 1].
124
125 The local trainer corrupts each per-agent target direction with a persistent
126 @c _local_bias (particle.py). With federation on, IFCA aggregates members'
127 models and the random per-agent biases cancel — the learned heading approaches
128 the true direction and this metric rises. Without federation, the bias persists
129 and the metric stays low. This is the headline ablation metric: CFL vs
130 individual learning under non-IID clients.
131
132 1.0 = perfect alignment with the unbiased ideal; 0.5 = orthogonal; 0.0 = anti-aligned.
133
134 @param particle A Particle with @c x, @c y, @c _target_x, @c _target_y, @c model.
135 @return float in [0, 1].
136 """
137 tx = particle._target_x - particle.x
138 ty = particle._target_y - particle.y
139 n = math.hypot(tx, ty)
140 if n == 0.0:
141 return 1.0
142 cos_sim = particle.model[0] * (tx / n) + particle.model[1] * (ty / n)
143 return float((cos_sim + 1.0) * 0.5)

150class SimLogger:
151 """@brief Records per-round statistics to disk and generates matplotlib plots.
152
153 Files written to `log_dir/run_<timestamp>/`:
154 rounds.csv — one row per CFL round, aggregate metrics
155 migrations.csv — one row per (round, src_cluster, dst_cluster) migration event

157 events.jsonl — split / merge / explosion events, one JSON object per line
158 sim_log.txt — human-readable mirror of terminal output
159 """

167 def __init__(
168 self,
169 log_dir: str = "logs",
170 tags: str = "",
171 cfl_enabled: bool = True,
172 attraction_enabled: bool = True,
173 run_name: Optional[str] = None,
174 ) -> None:
175 """@brief Create a run directory and open all CSV/JSONL/text writers.
176
177 @param log_dir Parent directory for run folders.
178 @param tags Optional suffix tag for the auto-generated run directory name (e.g. "cfl_emergent").
179 @param cfl_enabled Whether federation is on (recorded in the plot labels).
180 @param attraction_enabled Whether emergent forces are on (recorded in the plot labels).

185 self.run_dir = os.path.join(log_dir, run_name)

190 self.cfl_enabled = cfl_enabled
191 self.attraction_enabled = attraction_enabled
192 self._config_label = (
193 f"CFL: {'ON' if cfl_enabled else 'OFF'} | "
194 f"Emergent: {'ON' if attraction_enabled else 'OFF'}"
195 )
196
197 # ── open CSV writers ──────────────────────────────────────────────────
198 self._rounds_fh = open(os.path.join(self.run_dir, "rounds.csv"), "w", newline="")
199 self._migrations_fh = open(os.path.join(self.run_dir, "migrations.csv"), "w", newline="")
200 self._cluster_sizes_fh = open(os.path.join(self.run_dir, "cluster_sizes.csv"), "w", newline="")
201 self._events_fh = open(os.path.join(self.run_dir, "events.jsonl"), "w")
202 self._log_fh = open(os.path.join(self.run_dir, "sim_log.txt"), "w", encoding="utf-8")

206 fieldnames=[
207 "round", "num_clusters", "num_particles",
208 "inertia", "total_migrations", "event",
209 "avg_confidence", "avg_local_loss",
210 "avg_direction_precision", "avg_spatial_precision", "cluster_purity",
211 "avg_peer_alignment", "avg_obstacle_pressure",
212 "avg_drift_velocity", "avg_rounds_stable",
213 ],
214 )
215 self._rounds_writer.writeheader()
216
217 self._migrations_writer = csv.DictWriter(
218 self._migrations_fh,
219 fieldnames=["round", "src_cluster", "dst_cluster", "count"],
220 )
221 self._migrations_writer.writeheader()
222
223 self._cluster_sizes_writer = csv.DictWriter(
224 self._cluster_sizes_fh,
225 fieldnames=[
226 "round", "cluster_id", "size",
227 "avg_confidence", "avg_local_loss",
228 "avg_direction_precision", "avg_spatial_precision",
229 "avg_peer_alignment", "avg_obstacle_pressure",
230 "avg_drift_velocity", "avg_rounds_stable",
231 "model_divergence", "spatial_spread",
232 ],
233 )
234 self._cluster_sizes_writer.writeheader()
235
236 # ── in-memory history for plotting ───────────────────────────────────

238 self._history: List[Dict[str, Any]] = []

240 self._cluster_history: Dict[int, List[Dict[str, Any]]] = defaultdict(list)
241 # migration totals per round
242 self._migration_history: List[Tuple[int, int]] = [] # (round, total_migrations)
243 # events list for vertical markers
244 self._events: List[Dict[str, Any]] = []
245 # cumulative migration matrix: (src, dst) -> total count (in-memory, no CSV re-read)

248 self._log(f"SimLogger initialised — output dir: {self.run_dir}\n")

252 def log_round(
253 self,
254 round_num: int,
255 particles: list,
256 kmeans,
257 cluster_targets: list,
258 transfers: dict,
259 event: Optional[str],
260 num_clusters: int,
261 ) -> None:
262 """@brief Record one round to all CSVs and the in-memory plotting history.
263
264 Call once per round, right after run_cfl_round() returns.
265
266 @param round_num 1-based round index.
267 @param particles List of all Particle agents (read for aggregate metrics).
268 @param kmeans Fitted KMeans (read for @c .inertia_).
269 @param cluster_targets Current per-cluster (x, y) targets.
270 @param transfers dict (src, dst) -> migration count for this round.
271 @param event None, 'split', 'merge', or 'explosion' for this round.
272 @param num_clusters Current number of clusters.
273 @return None.
274 """
275
276 inertia = float(kmeans.inertia_) if hasattr(kmeans, "inertia_") else 0.0
277
278 # ── aggregate particle metrics ────────────────────────────────────────
279 models = np.array([p.model for p in particles])
280 avg_conf = float(np.mean(models[:, 2]))
281 avg_loss = float(np.mean(models[:, 6]))
282 avg_peer = float(np.mean(models[:, 4]))
283 avg_obs = float(np.mean(models[:, 3]))
284 avg_drift = float(np.mean(models[:, 7]))
285 avg_stable = float(np.mean(models[:, 5]))
286
287 # Direction precision: per-agent cosine alignment of learned heading to
288 # the *true* (unbiased) target direction, in [0, 1]. Headline ablation
289 # metric — rises under CFL (federation averages out non-IID biases),
290 # stays low under individual learning.
291 dir_precisions = [_direction_precision(p) for p in particles]
292 avg_dir_prec = float(np.mean(dir_precisions)) if dir_precisions else 0.0
293
294 # Spatial precision: 1 - distance/sim_diagonal. Complements direction
295 # precision — distinguishes "model points the right way" from "particle
296 # actually arrives at the target". Emergent attraction/repulsion forces
297 # affect this without touching direction_precision.
298 spatial_precisions = [_spatial_precision(p, _MAX_DIST) for p in particles]
299 avg_spatial_prec = float(np.mean(spatial_precisions)) if spatial_precisions else 0.0
300
301 # Cluster purity: how well predicted cluster_id matches the latent
302 # _true_cluster_id labels. None when no ground truth is set (legacy).
303 purity = _cluster_purity(particles)
304
305 total_mig = sum(transfers.values()) if transfers else 0
306
307 # ── write rounds.csv ─────────────────────────────────────────────────
308 row = {
309 "round": round_num,
310 "num_clusters": num_clusters,
311 "num_particles": len(particles),
312 "inertia": round(inertia, 4),
313 "total_migrations": total_mig,
314 "event": event or "",
315 "avg_confidence": round(avg_conf, 4),
316 "avg_local_loss": round(avg_loss, 4),
317 "avg_direction_precision": round(avg_dir_prec, 4),
318 "avg_spatial_precision": round(avg_spatial_prec, 4),
319 "cluster_purity": None if purity is None else round(purity, 4),
320 "avg_peer_alignment": round(avg_peer, 4),
321 "avg_obstacle_pressure": round(avg_obs, 4),
322 "avg_drift_velocity": round(avg_drift, 4),
323 "avg_rounds_stable": round(avg_stable, 4),
324 }
325 self._rounds_writer.writerow(row)
326 self._rounds_fh.flush()
327 self._history.append(row)
328 self._migration_history.append((round_num, total_mig))
329
330 # ── write migrations.csv ──────────────────────────────────────────────
331 for (src, dst), count in (transfers or {}).items():
332 self._migrations_writer.writerow({
333 "round": round_num, "src_cluster": src,
334 "dst_cluster": dst, "count": count,
335 })
336 if src >= 0:
337 self._mig_matrix[(src, dst)] += count
338 self._migrations_fh.flush()
339
340 # ── write cluster_sizes.csv ───────────────────────────────────────────
341 cluster_buckets: Dict[int, list] = defaultdict(list)
342 for p in particles:
343 cluster_buckets[p.cluster_id].append(p)
344
345 cluster_rows = {}
346 for cid in range(num_clusters):
347 members = cluster_buckets.get(cid, [])
348 if not members:
349 continue
350 ms = np.array([p.model for p in members])
351
352 # model divergence: mean std across key dimensions (dir, confidence, loss)
353 model_div = float(np.mean(np.std(ms[:, [0, 1, 2, 6]], axis=0))) if len(members) > 1 else 0.0
354
355 # spatial spread: mean distance from cluster centroid
356 xs = np.array([p.x for p in members])
357 ys = np.array([p.y for p in members])
358 cx, cy = xs.mean(), ys.mean()
359 spatial_spread = float(np.mean(np.sqrt((xs - cx) ** 2 + (ys - cy) ** 2)))
360
361 cluster_dir_prec = float(np.mean([_direction_precision(p) for p in members]))
362 cluster_spatial_prec = float(np.mean([_spatial_precision(p, _MAX_DIST) for p in members]))
363
364 cs_row = {
365 "round": round_num,
366 "cluster_id": cid,
367 "size": len(members),
368 "avg_confidence": round(float(np.mean(ms[:, 2])), 4),
369 "avg_local_loss": round(float(np.mean(ms[:, 6])), 4),
370 "avg_direction_precision": round(cluster_dir_prec, 4),
371 "avg_spatial_precision": round(cluster_spatial_prec, 4),
372 "avg_peer_alignment": round(float(np.mean(ms[:, 4])), 4),
373 "avg_obstacle_pressure":round(float(np.mean(ms[:, 3])), 4),
374 "avg_drift_velocity": round(float(np.mean(ms[:, 7])), 4),
375 "avg_rounds_stable": round(float(np.mean(ms[:, 5])), 4),
376 "model_divergence": round(model_div, 4),
377 "spatial_spread": round(spatial_spread, 1),
378 }
379 self._cluster_sizes_writer.writerow(cs_row)
380 self._cluster_history[cid].append(cs_row)
381 cluster_rows[cid] = cs_row
382 self._cluster_sizes_fh.flush()
383
384 # ── write events.jsonl ────────────────────────────────────────────────
385 if event:
386 ev = {"round": round_num, "type": event, "num_clusters": num_clusters}
387 self._events_fh.write(json.dumps(ev) + "\n")
388 self._events_fh.flush()
389 self._events.append(ev)
390
391 self._write_terminal_log(
392 round_num, num_clusters, inertia, avg_conf, avg_loss,
393 avg_peer, avg_dir_prec, avg_spatial_prec, purity,
394 total_mig, event, transfers, cluster_rows,
395 )
