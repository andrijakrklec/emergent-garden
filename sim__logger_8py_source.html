66class SimLogger:
67 """@brief Records per-round statistics to disk and generates matplotlib plots.
68
69 Files written to `log_dir/run_<timestamp>/`:
70 rounds.csv — one row per CFL round, aggregate metrics
71 migrations.csv — one row per (round, src_cluster, dst_cluster) migration event
72 cluster_sizes.csv — one row per (round, cluster_id) with size + model stats
73 events.jsonl — split / merge / explosion events, one JSON object per line
74 sim_log.txt — human-readable mirror of terminal output
75 """
76

83 def __init__(
84 self,
85 log_dir: str = "logs",
86 tags: str = "",
87 cfl_enabled: bool = True,
88 attraction_enabled: bool = True,
89 ) -> None:
90 """@brief Create a timestamped run directory and open all CSV/JSONL/text writers.
91
92 @param log_dir Parent directory for run folders.
93 @param tags Optional suffix tag for the run directory name (e.g. "cfl_emergent").
94 @param cfl_enabled Whether federation is on (recorded in the plot labels).
95 @param attraction_enabled Whether emergent forces are on (recorded in the plot labels).
96 """
97 suffix = f"_{tags}" if tags else ""
98 self.run_dir = os.path.join(log_dir, f"run_{_ts()}{suffix}")
99 _ensure_dir(self.run_dir)
100 self.cfl_enabled = cfl_enabled
101 self.attraction_enabled = attraction_enabled
102 self._config_label = (
103 f"CFL: {'ON' if cfl_enabled else 'OFF'} | "
104 f"Emergent: {'ON' if attraction_enabled else 'OFF'}"
105 )
106
107 # ── open CSV writers ──────────────────────────────────────────────────
108 self._rounds_fh = open(os.path.join(self.run_dir, "rounds.csv"), "w", newline="")
109 self._migrations_fh = open(os.path.join(self.run_dir, "migrations.csv"), "w", newline="")
110 self._cluster_sizes_fh = open(os.path.join(self.run_dir, "cluster_sizes.csv"), "w", newline="")
111 self._events_fh = open(os.path.join(self.run_dir, "events.jsonl"), "w")
112 self._log_fh = open(os.path.join(self.run_dir, "sim_log.txt"), "w", encoding="utf-8")
113
114 self._rounds_writer = csv.DictWriter(
115 self._rounds_fh,
116 fieldnames=[
117 "round", "num_clusters", "num_particles",
118 "inertia", "total_migrations", "event",
119 "avg_confidence", "avg_local_loss",
120 "avg_peer_alignment", "avg_obstacle_pressure",
121 "avg_drift_velocity", "avg_rounds_stable",
122 ],
123 )
124 self._rounds_writer.writeheader()
125
126 self._migrations_writer = csv.DictWriter(
127 self._migrations_fh,
128 fieldnames=["round", "src_cluster", "dst_cluster", "count"],
129 )
130 self._migrations_writer.writeheader()
131
132 self._cluster_sizes_writer = csv.DictWriter(
133 self._cluster_sizes_fh,
134 fieldnames=[
135 "round", "cluster_id", "size",
136 "avg_confidence", "avg_local_loss",
137 "avg_peer_alignment", "avg_obstacle_pressure",
138 "avg_drift_velocity", "avg_rounds_stable",
139 "model_divergence", "spatial_spread",
140 ],
141 )
142 self._cluster_sizes_writer.writeheader()
143
144 # ── in-memory history for plotting ───────────────────────────────────
145 # round-level
146 self._history: List[Dict[str, Any]] = []
147 # per-cluster history: {cluster_id: [{"round":…, "size":…, …}, …]}
148 self._cluster_history: Dict[int, List[Dict[str, Any]]] = defaultdict(list)
149 # migration totals per round
150 self._migration_history: List[Tuple[int, int]] = [] # (round, total_migrations)
151 # events list for vertical markers
152 self._events: List[Dict[str, Any]] = []
153 # cumulative migration matrix: (src, dst) -> total count (in-memory, no CSV re-read)
154 self._mig_matrix: Dict[Tuple[int, int], int] = defaultdict(int)
155
156 self._log(f"SimLogger initialised — output dir: {self.run_dir}\n")

160 def log_round(
161 self,
162 round_num: int,
163 particles: list,
164 kmeans,
165 cluster_targets: list,
166 transfers: dict,
167 event: Optional[str],
168 num_clusters: int,
169 ) -> None:
170 """@brief Record one round to all CSVs and the in-memory plotting history.
171
172 Call once per round, right after run_cfl_round() returns.
173
174 @param round_num 1-based round index.
175 @param particles List of all Particle agents (read for aggregate metrics).
176 @param kmeans Fitted KMeans (read for @c .inertia_).
177 @param cluster_targets Current per-cluster (x, y) targets.
178 @param transfers dict (src, dst) -> migration count for this round.
179 @param event None, 'split', 'merge', or 'explosion' for this round.
180 @param num_clusters Current number of clusters.
181 @return None.
182 """
183
184 inertia = float(kmeans.inertia_) if hasattr(kmeans, "inertia_") else 0.0
185
186 # ── aggregate particle metrics ────────────────────────────────────────
187 models = np.array([p.model for p in particles])
188 avg_conf = float(np.mean(models[:, 2]))
189 avg_loss = float(np.mean(models[:, 6]))
190 avg_peer = float(np.mean(models[:, 4]))
191 avg_obs = float(np.mean(models[:, 3]))
192 avg_drift = float(np.mean(models[:, 7]))
193 avg_stable = float(np.mean(models[:, 5]))
194
195 total_mig = sum(transfers.values()) if transfers else 0
196
197 # ── write rounds.csv ─────────────────────────────────────────────────
198 row = {
199 "round": round_num,
200 "num_clusters": num_clusters,
201 "num_particles": len(particles),
202 "inertia": round(inertia, 4),
203 "total_migrations": total_mig,
204 "event": event or "",
205 "avg_confidence": round(avg_conf, 4),
206 "avg_local_loss": round(avg_loss, 4),
207 "avg_peer_alignment": round(avg_peer, 4),
208 "avg_obstacle_pressure": round(avg_obs, 4),
209 "avg_drift_velocity": round(avg_drift, 4),
210 "avg_rounds_stable": round(avg_stable, 4),
211 }
212 self._rounds_writer.writerow(row)
213 self._rounds_fh.flush()
214 self._history.append(row)
215 self._migration_history.append((round_num, total_mig))
216
217 # ── write migrations.csv ──────────────────────────────────────────────
218 for (src, dst), count in (transfers or {}).items():
219 self._migrations_writer.writerow({
220 "round": round_num, "src_cluster": src,
221 "dst_cluster": dst, "count": count,
222 })
223 if src >= 0:
224 self._mig_matrix[(src, dst)] += count
225 self._migrations_fh.flush()
226
227 # ── write cluster_sizes.csv ───────────────────────────────────────────
228 cluster_buckets: Dict[int, list] = defaultdict(list)
229 for p in particles:
230 cluster_buckets[p.cluster_id].append(p)
231
232 cluster_rows = {}
233 for cid in range(num_clusters):
234 members = cluster_buckets.get(cid, [])
235 if not members:
236 continue
237 ms = np.array([p.model for p in members])
238
239 # model divergence: mean std across key dimensions (dir, confidence, loss)
240 model_div = float(np.mean(np.std(ms[:, [0, 1, 2, 6]], axis=0))) if len(members) > 1 else 0.0
241
242 # spatial spread: mean distance from cluster centroid
243 xs = np.array([p.x for p in members])
244 ys = np.array([p.y for p in members])
245 cx, cy = xs.mean(), ys.mean()
246 spatial_spread = float(np.mean(np.sqrt((xs - cx) ** 2 + (ys - cy) ** 2)))
247
248 cs_row = {
249 "round": round_num,
250 "cluster_id": cid,
251 "size": len(members),
252 "avg_confidence": round(float(np.mean(ms[:, 2])), 4),
253 "avg_local_loss": round(float(np.mean(ms[:, 6])), 4),
254 "avg_peer_alignment": round(float(np.mean(ms[:, 4])), 4),
255 "avg_obstacle_pressure":round(float(np.mean(ms[:, 3])), 4),
256 "avg_drift_velocity": round(float(np.mean(ms[:, 7])), 4),
257 "avg_rounds_stable": round(float(np.mean(ms[:, 5])), 4),
258 "model_divergence": round(model_div, 4),
259 "spatial_spread": round(spatial_spread, 1),
260 }
261 self._cluster_sizes_writer.writerow(cs_row)
262 self._cluster_history[cid].append(cs_row)
263 cluster_rows[cid] = cs_row
264 self._cluster_sizes_fh.flush()
265
266 # ── write events.jsonl ────────────────────────────────────────────────
267 if event:
268 ev = {"round": round_num, "type": event, "num_clusters": num_clusters}
269 self._events_fh.write(json.dumps(ev) + "\n")
270 self._events_fh.flush()
271 self._events.append(ev)
272
273 self._write_terminal_log(
274 round_num, num_clusters, inertia, avg_conf, avg_loss,
275 avg_peer, total_mig, event, transfers, cluster_rows,
276 )
