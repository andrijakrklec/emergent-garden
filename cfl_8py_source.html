28def compute_cluster_stats(particles, n_clusters):
29 """@brief Aggregate per-cluster health used by the split/merge logic and logger.
30
31 @param particles List of all Particle agents.
32 @param n_clusters Number of clusters to report on (ids 0..n_clusters-1).
33 @return dict cluster_id -> {size, avg_loss, avg_confidence, mean_model}, where
34 @c mean_model is the element-wise mean 8-D model (None for empty clusters).
35 """
36 stats = {}
37 for cid in range(n_clusters):
38 members = [p for p in particles if p.cluster_id == cid]
39 if not members:
40 stats[cid] = {'size': 0, 'avg_loss': 0.0,
41 'avg_confidence': 0.0, 'mean_model': None}
42 continue
43 avg_loss = sum(p.model[6] for p in members) / len(members)
44 avg_conf = sum(p.model[2] for p in members) / len(members)
45 mean_model = sum(p.model for p in members) / len(members)
46 stats[cid] = {
47 'size': len(members),
48 'avg_loss': avg_loss,
49 'avg_confidence': avg_conf,
50 'mean_model': mean_model,
51 }
52 return stats

54def _direction_coherence(members):
55 """@brief Directional coherence of a cluster — the IFCA split signal.
56
57 Mean cosine similarity of members' headings to the cluster's own mean
58 heading: 1.0 = perfectly aligned, 0 = chaotic, -1 = anti-aligned. Under IFCA
59 avg_loss is pinned near the local minimum after assignment, so loss cannot
60 reveal an incoherent cluster — directional disagreement among members can.
61
62 @param members List of Particle agents in one cluster.
63 @return float coherence in [-1, 1] (1.0 when fewer than two members).
64 """
65 if len(members) < 2:
66 return 1.0
67 dirs = np.array([p.model[0:2] for p in members])
68 mean_dir = dirs.mean(axis=0)
69 norm = np.linalg.norm(mean_dir)
70 if norm == 0:
71 return 0.0
72 mean_dir = mean_dir / norm
73 return float((dirs @ mean_dir).mean())

76def _ifca_score(particle, target, theta, max_dist):
77 """@brief IFCA loss for evaluating cluster k on a single agent (lower is better).
78
79 Mirrors the per-agent loss from local_train() (model[6]) but evaluated
80 under cluster k's *broadcast* model theta and cluster k's target. This is the
81 score each IFCA client computes locally for every candidate model in order to
82 pick its cluster (argmin over k).
83
84 @param particle The agent being evaluated.
85 @param target (x, y) spatial target of cluster k.
86 @param theta Cluster k's broadcast 8-D model (its heading is theta[0:2]); may be None.
87 @param max_dist Diagonal of the simulation area, used to normalize distance.
88 @return float loss = 0.5 * geometric_loss + 0.5 * directional_loss.
89 """
90 dx = target[0] - particle.x
91 dy = target[1] - particle.y
92 dist = math.hypot(dx, dy)
93 geo_loss = min(dist / max_dist, 1.0)
94
95 if dist > 0 and theta is not None:
96 true_dx, true_dy = dx / dist, dy / dist
97 # Alignment of θ_k with the true direction from particle to target_k.
98 alignment = float(theta[0] * true_dx + theta[1] * true_dy) # -1..1
99 dir_loss = (1.0 - alignment) / 2.0
100 else:
101 dir_loss = 0.5 # no info → neutral
102
103 return 0.5 * geo_loss + 0.5 * dir_loss

106def _compute_cluster_models(particles, n_clusters):
107 """@brief Confidence-weighted mean model per cluster — the broadcast theta_k.
108
109 Each cluster's model is the mean of its members' 8-D models weighted by
110 confidence (model[2]), with the heading [0:2] re-normalized to a unit vector.
111
112 @param particles List of all Particle agents.
113 @param n_clusters Number of clusters.
114 @return list of length @p n_clusters; each entry is a length-8 numpy array, or
115 None for an empty cluster.
116 """
117 cluster_models = [None] * n_clusters
118 for k in range(n_clusters):
119 members = [p for p in particles if p.cluster_id == k]
120 if not members:
121 continue
122 weights = np.array([p.model[2] for p in members])
123 weights /= weights.sum() + 1e-8
124 theta = sum(w * p.model for w, p in zip(weights, members))
125 norm = np.linalg.norm(theta[0:2])
126 if norm > 0:
127 theta[0:2] = theta[0:2] / norm
128 cluster_models[k] = theta
129 return cluster_models

132def _merge_clusters(particles, n, stats, cluster_targets, cluster_colors, cluster_ages, transfers):
133 """@brief One merge step: absorb a tiny or redundant cluster, if any.
134
135 Two paths, tried in order: (1) absorb any cluster smaller than
136 @c min_cluster_size into its spatially nearest neighbor; (2) merge the most
137 redundant mature pair (high heading similarity OR near-coincident targets).
138
139 @param particles List of all Particle agents; cluster_id mutated in place on a merge.
140 @param n Current number of clusters.
141 @param stats Per-cluster stats from compute_cluster_stats().
142 @param cluster_targets List of (x, y) cluster targets.
143 @param cluster_colors dict cluster_id -> RGB (key -1 reserved for unassigned).
144 @param cluster_ages dict cluster_id -> age in rounds.
145 @param transfers dict (src, dst) -> migration count, threaded into the return tuple.
146 @return The 8-tuple run_cfl_round returns on a merge (event = ('merge', dropped_id)),
147 or None when no merge fires.
148 """
149 if n <= cfl_params.get_min_clusters():
150 return None
151
152 min_cluster_size = cfl_params.get_min_cluster_size()
153 merge_pair = None
154
155 # Path 1: Absorb tiny clusters. A cluster with fewer than min_cluster_size
156 # members is effectively dead under IFCA (its θ_k is too noisy) — fold
157 # it into the cluster whose target is nearest, regardless of age.
158 tiny = [cid for cid in range(n) if stats[cid]['size'] < min_cluster_size]
159 for tcid in tiny:
160 tx, ty = cluster_targets[tcid]
161 other = [cid for cid in range(n) if cid != tcid and stats[cid]['size'] > 0]
162 if not other:
163 continue
164 nearest = min(
165 other,
166 key=lambda cid: math.hypot(cluster_targets[cid][0] - tx,
167 cluster_targets[cid][1] - ty),
168 )
169 merge_pair = (min(nearest, tcid), max(nearest, tcid))
170 print(f" > MERGE-tiny: cluster {tcid} (size={stats[tcid]['size']}) absorbed into {nearest}")
171 break
172
173 # Path 2: Standard similarity-based merge for mature, redundant clusters.
174 if merge_pair is None:
175 best_score, candidate = -float('inf'), None
176 for a in range(n):
177 for b in range(a + 1, n):
178 ma = stats[a]['mean_model']
179 mb = stats[b]['mean_model']
180 if ma is None or mb is None:
181 continue
182 maturity_rounds = cfl_params.get_blend_maturity_rounds()
183 if (cluster_ages.get(a, 0) < maturity_rounds or
184 cluster_ages.get(b, 0) < maturity_rounds):
185 continue
186
187 sim = float(np.dot(ma[0:2], mb[0:2]))
188 # Spatial-target component: 1.0 when targets coincide,
189 # 0.0 when they're the merge-target distance or farther apart.
190 tdist = math.hypot(
191 cluster_targets[a][0] - cluster_targets[b][0],
192 cluster_targets[a][1] - cluster_targets[b][1],
193 )
194 target_score = max(0.0, 1.0 - tdist / cfl_params.get_merge_target_distance())
195 # Combined score: high direction sim OR very close targets.
196 score = max(sim, target_score)
197 if score > best_score:
198 best_score, candidate = score, (a, b)
199
200 if candidate and best_score >= cfl_params.get_merge_similarity_threshold():
201 merge_pair = candidate
202 print(f" > MERGE-redundant: {candidate} score={best_score:.3f}")
203
204 if not merge_pair:
205 return None
206
207 keep, drop = merge_pair
208 for p in particles:
209 if p.cluster_id == drop:
210 p.cluster_id = keep
211 p._prev_cluster_id = -1 # force stability reset next round
212 p._stable_rounds = 0
213 for p in particles:
214 if p.cluster_id > drop:
215 p.cluster_id -= 1
216 p._prev_cluster_id = p.cluster_id
217
218 new_targets = [t for i, t in enumerate(cluster_targets) if i != drop]
219 new_colors = {(i if i < drop else i - 1): c
220 for i, c in cluster_colors.items() if i != drop and i != -1}
221 new_colors[-1] = cluster_colors[-1]
222
223 # Remap ages, drop the merged cluster
224 new_ages = {}
225 for cid, age in cluster_ages.items():
226 if cid == drop:
227 continue
228 new_cid = cid if cid < drop else cid - 1
229 new_ages[new_cid] = age
230 # The surviving cluster inherits the older age
231 new_ages[keep if keep < drop else keep - 1] = max(
232 cluster_ages.get(keep, 0), cluster_ages.get(drop, 0)
233 )
234
235 new_n = n - 1
236 new_kmeans = KMeans(n_clusters=new_n, n_init=10, random_state=0)
237
238 max_target_idx = len(new_targets) - 1
239 for p in particles:
240 p.target_idx = min(p.target_idx, max_target_idx)
241
242 return transfers, new_kmeans, new_targets, new_colors, new_ages, new_n, ('merge', drop), 10

245def _split_clusters(particles, n, stats, cluster_targets, cluster_colors, cluster_ages, transfers):
246 """@brief One split step: carve an incoherent cluster in two, if one qualifies.
247
248 Under IFCA the split signal is directional incoherence among members
249 (avg_loss is mechanically pinned at the local min after assignment). The
250 worst-scoring eligible cluster is split via 2-means on member heading
251 vectors, falling back to a spatial-x split when 2-means is too unbalanced.
252
253 @param particles List of all Particle agents; cluster_id mutated in place on a split.
254 @param n Current number of clusters.
255 @param stats Per-cluster stats from compute_cluster_stats().
256 @param cluster_targets List of (x, y) cluster targets.
257 @param cluster_colors dict cluster_id -> RGB (key -1 reserved for unassigned).
258 @param cluster_ages dict cluster_id -> age in rounds.
259 @param transfers dict (src, dst) -> migration count, threaded into the return tuple.
260 @return The 8-tuple run_cfl_round returns on a split (event = 'split'), or None
261 when no split fires.
262 """
263 if n >= cfl_params.get_max_clusters():
264 return None
265
266 min_cluster_size = cfl_params.get_min_cluster_size()
267 eligible = [cid for cid in range(n)
268 if stats[cid]['size'] >= min_cluster_size * 2]
269
270 worst_cid, worst_score = None, 0.0
271 for cid in eligible:
272 members = [p for p in particles if p.cluster_id == cid]
273 coherence = _direction_coherence(members)
274 avg_loss = stats[cid]['avg_loss']
275
276 # Combined "wants to split" score: how much each criterion exceeds
277 # its threshold. Either alone can trigger.
278 coh_excess = max(0.0, cfl_params.get_split_coherence_threshold() - coherence)
279 loss_excess = max(0.0, avg_loss - cfl_params.get_split_loss_threshold())
280 score = coh_excess + loss_excess
281 if score > worst_score:
282 worst_score, worst_cid = score, cid
283
284 if worst_cid is None or worst_score <= 0.0:
285 return None
286
287 members = [p for p in particles if p.cluster_id == worst_cid]
288 members_coh = _direction_coherence(members)
289 members_avg = stats[worst_cid]['avg_loss']
290 print(f" > SPLIT-trigger: cluster {worst_cid} "
291 f"coherence={members_coh:.2f} avg_loss={members_avg:.2f}")
292
293 # Split along the direction-disagreement axis: 2-means on member
294 # heading vectors. Members in one heading group form the new cluster.
295 # Falls back to spatial-x split when 2-means is too unbalanced.
296 new_members = None
297 if len(members) >= min_cluster_size * 2:
298 try:
299 dirs = np.array([p.model[0:2] for p in members])
300 km2 = KMeans(n_clusters=2, n_init=3, random_state=0)
301 labels = km2.fit_predict(dirs)
302 group_a = [m for m, l in zip(members, labels) if l == 0]
303 group_b = [m for m, l in zip(members, labels) if l == 1]
304 if len(group_a) >= min_cluster_size and len(group_b) >= min_cluster_size:
305 # Smaller group becomes the new cluster.
306 new_members = group_b if len(group_b) <= len(group_a) else group_a
307 except Exception:
308 pass
309
310 if new_members is None:
311 members_sorted = sorted(members, key=lambda p: p.x)
312 split_count = max(min_cluster_size, len(members_sorted) // 2)
313 new_members = members_sorted[:split_count]
314
315 new_cid = n
316 for p in new_members:
317 p.cluster_id = new_cid
318 p._prev_cluster_id = -1
319 p._stable_rounds = 0
320
321 ox, oy = cluster_targets[worst_cid]
322 new_targets = cluster_targets + [(
323 int(np.clip(ox + random.randint(-150, 150), 50, SIM_WIDTH - 50)),
324 int(np.clip(oy + random.randint(-150, 150), 50, SIM_HEIGHT - 50))
325 )]
326 used = set(cluster_colors.values()) - {cluster_colors[-1]}
327 new_colors = dict(cluster_colors)
328 new_colors[new_cid] = _pick_unused_color(used)
329
330 # New cluster starts at age 0, parent keeps its age
331 new_ages = dict(cluster_ages)
332 new_ages[new_cid] = 0
333
334 new_n = n + 1
335 new_kmeans = KMeans(n_clusters=new_n, n_init=10, random_state=0)
336
337 new_cid = n
338 for p in particles:
339 if p.cluster_id == new_cid:
340 p.target_idx = len(new_targets) - 1
341
342 return transfers, new_kmeans, new_targets, new_colors, new_ages, new_n, 'split', 10

345def run_cfl_round(particles, kmeans_model, cluster_targets, cluster_colors, cluster_ages, cooldown_counter):
346 """@brief One IFCA-style clustered federated learning round (Ghosh et al. 2020).
347
348 Steps: (1) broadcast a confidence-weighted model theta_k per cluster; (2) each
349 agent re-assigns to the cluster whose theta_k gives the lowest loss (argmin),
350 subject to migration hysteresis; (3) aggregate members toward theta_k with an
351 age-adaptive blend; (4) rescue stuck stragglers; (5) after a cooldown, possibly
352 @b merge redundant/tiny clusters or @b split an incoherent one. A fresh KMeans
353 is fit at the end purely to preserve the @c .n_clusters / @c .inertia_ contract
354 that the logger reads — IFCA itself does not need it.
355
356 @param particles List of all Particle agents; cluster_id and model mutated in place.
357 @param kmeans_model Previous KMeans (read for @c .n_clusters; re-fit and returned).
358 @param cluster_targets List of (x, y) spatial targets, one per cluster.
359 @param cluster_colors dict cluster_id -> RGB (key -1 reserved for unassigned).
360 @param cluster_ages dict cluster_id -> age in rounds (drives the blend ratio).
361 @param cooldown_counter Rounds remaining before another split/merge may fire.
362 @return 8-tuple (transfers, kmeans_model, cluster_targets, cluster_colors,
363 cluster_ages, num_clusters, event, cooldown_counter), where @c transfers maps
364 (src, dst) -> migration count and @c event is None, 'split', or ('merge', dropped_id).
365 """
366 if not particles:
367 return {}, kmeans_model, cluster_targets, cluster_colors, cluster_ages, kmeans_model.n_clusters, None, cooldown_counter
368
369 n = kmeans_model.n_clusters
370 old_ids = [p.cluster_id for p in particles]
371
372 # --- IFCA broadcast: build θ_k from current cluster membership ---
373 cluster_models = _compute_cluster_models(particles, n)
374 max_dist = math.hypot(SIM_DIM[0], SIM_DIM[1])
375
376 # Fallback for clusters that are currently empty: seed θ_k with a
377 # unit vector pointing from the simulation center toward target_k,
378 # so the loss is well-defined and the cluster can still attract members.
379 cx, cy = SIM_DIM[0] / 2.0, SIM_DIM[1] / 2.0
380 for k in range(n):
381 if cluster_models[k] is None and k < len(cluster_targets):
382 tx, ty = cluster_targets[k]
383 dx, dy = tx - cx, ty - cy
384 d = math.hypot(dx, dy)
385 theta = np.zeros(8, dtype=np.float64)
386 if d > 0:
387 theta[0], theta[1] = dx / d, dy / d
388 theta[2] = 0.5 # middling confidence
389 cluster_models[k] = theta
390
391 # --- IFCA assignment: each particle picks argmin loss across θ_k ---
392 transfers = defaultdict(int)
393 for i, p in enumerate(particles):
394 old_id = old_ids[i]
395
396 losses = [
397 _ifca_score(p, cluster_targets[k], cluster_models[k], max_dist)
398 for k in range(n)
399 ]
400 new_id = int(np.argmin(losses))
401
402 # Hysteresis: only switch if the chosen cluster's loss is meaningfully
403 # below the current cluster's loss. Stops particles flipping between
404 # nearly-equivalent clusters when geometry barely favors one.
405 if 0 <= old_id < n and old_id != new_id:
406 if losses[new_id] >= losses[old_id] * (1.0 - cfl_params.get_migration_hysteresis()):
407 new_id = old_id
408
409 if old_id != new_id:
410 transfers[(old_id, new_id)] += 1
411 p.cluster_id = new_id
412
413 # --- Aggregation: recompute θ_k from new assignments and blend ---
414 aggregated_models = _compute_cluster_models(particles, n)
415 for cid in range(n):
416 aggregated = aggregated_models[cid]
417 if aggregated is None:
418 continue
419 members = [p for p in particles if p.cluster_id == cid]
420
421 # Blend ratio depends on cluster age (newborn clusters keep more
422 # local state so they don't immediately re-merge with the parent).
423 maturity_rounds = cfl_params.get_blend_maturity_rounds()
424 age = cluster_ages.get(cid, maturity_rounds)
425 t = min(age / maturity_rounds, 1.0) # 0.0 = newborn, 1.0 = mature
426 local_new = cfl_params.get_blend_local_new()
427 local_weight = local_new + t * (cfl_params.get_blend_local_mature() - local_new)
428 global_weight = 1.0 - local_weight
429
430 for p in members:
431 p.model = local_weight * p.model + global_weight * aggregated
432 p.model[2] = np.clip(p.model[2], 0.01, 1.0)
433 p.model[3:5] = np.clip(p.model[3:5], 0.0, 1.0)
434
435 # --- Re-fit KMeans purely for the .inertia_ / .n_clusters contract ---
436 # IFCA itself doesn't need this; downstream logging/printing reads it.
437 try:
438 identity_slice = cfl_params.get_identity_slice()
439 identity_models = np.array([p.model[identity_slice] for p in particles])
440 init_centroids = np.array([
441 np.mean([p.model[identity_slice] for p in particles if p.cluster_id == cid], axis=0)
442 if any(p.cluster_id == cid for p in particles)
443 else identity_models[np.random.randint(len(identity_models))]
444 for cid in range(n)
445 ])
446 warm_kmeans = KMeans(n_clusters=n, init=init_centroids, n_init=1, random_state=0)
447 warm_kmeans.fit(identity_models)
448 kmeans_model = warm_kmeans
449 except Exception:
450 pass
451
452 # Stable round counter
453 for p in particles:
454 if p.cluster_id == p._prev_cluster_id:
455 p._stable_rounds = min(p._stable_rounds + 1, 50)
456 else:
457 p._stable_rounds = 0
458 p._prev_cluster_id = p.cluster_id
459 p.model[5] = p._stable_rounds / 50.0
460
461 # Increment age for all existing clusters
462 for cid in range(n):
463 cluster_ages[cid] = cluster_ages.get(cid, 0) + 1
464
465 # Straggler rescue: particles with high loss and near-zero drift are stuck.
466 # Re-assign them to the spatially nearest cluster target every round
467 # (runs even during cooldown so stuck particles aren't left orphaned).
468 straggler_loss = cfl_params.get_straggler_loss()
469 straggler_drift = cfl_params.get_straggler_drift()
470 for p in particles:
471 if p.model[6] > straggler_loss and p.model[7] < straggler_drift:
472 nearest = min(
473 range(len(cluster_targets)),
474 key=lambda i: math.hypot(p.x - cluster_targets[i][0],
475 p.y - cluster_targets[i][1])
476 )
477 if nearest != p.cluster_id:
478 p.cluster_id = nearest
479 p.target_idx = nearest
480 p._prev_cluster_id = -1
481 p._stable_rounds = 0
482
483 # --- Restructuring ---
484 if cooldown_counter > 0:
485 return transfers, kmeans_model, cluster_targets, cluster_colors, cluster_ages, n, None, cooldown_counter - 1
486
487 stats = compute_cluster_stats(particles, n)
488
489 # MERGE
490 merged = _merge_clusters(particles, n, stats, cluster_targets,
491 cluster_colors, cluster_ages, transfers)
492 if merged is not None:
493 return merged
494
495 # SPLIT
496 split = _split_clusters(particles, n, stats, cluster_targets,
497 cluster_colors, cluster_ages, transfers)
498 if split is not None:
499 return split
500
501 return transfers, kmeans_model, cluster_targets, cluster_colors, cluster_ages, n, None, 0

504 num: int,
505 c: tuple,
506 frame: Tuple[Tuple[int, int], Tuple[int, int]],
507 target_idx: int,
508 true_cluster_id: int,

510 ) -> List[Particle]:
511 """@brief Instantiate a group of agents spawned uniformly within a frame.
