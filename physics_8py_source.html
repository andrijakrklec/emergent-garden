24def gravity_force(g, distance, count, power=1):
25 """@brief Standard inverse-distance force scalar: g / (distance**power * count).
26
27 Multiply the returned scalar by the (dx, dy) component delta between the two
28 bodies to get the force vector. @p count divides the force so denser swarms
29 don't blow up. Guards a zero denominator.
30
31 @param g Interaction coefficient (negative attracts, positive repels).
32 @param distance Center-to-center distance between the two bodies.
33 @param count Number of interacting bodies (normalizes the total force).
34 @param power Exponent on distance (1 = inverse-linear falloff).
35 @return float force scalar.
36 """
37 denom = (distance ** power) * count
38 if denom == 0:
39 denom = 0.001
40 return g * (1.0 / denom)

52def interaction_coefficient(cluster_a, cluster_b, g_attract, g_repel, rules=None):
53 """@brief Resolve the attract/repel coefficient g for a pair of bodies.
54
55 Same-cluster pairs attract (@p g_attract), cross-cluster pairs repel
56 (@p g_repel), unless an explicit @p rules entry overrides the pair. Bodies
57 with no cluster (id -1) don't interact (returns 0).
58
59 @param cluster_a Cluster id of the first body (-1 = unassigned).
60 @param cluster_b Cluster id of the second body (-1 = unassigned).
61 @param g_attract Default same-cluster coefficient (negative attracts).
62 @param g_repel Default cross-cluster coefficient (positive repels).
63 @param rules Optional dict (min(a, b), max(a, b)) -> coefficient override.
64 @return float coefficient g (0.0 when either body is unassigned).
65 """
66 if cluster_a == -1 or cluster_b == -1:
67 return 0.0
68 if rules is not None:
69 key = (min(cluster_a, cluster_b), max(cluster_a, cluster_b))
70 default = g_attract if cluster_a == cluster_b else g_repel
71 return rules.get(key, default)
72 if cluster_a == cluster_b:
73 return g_attract
74 return g_repel

77def pair_force(distance, coefficient, count, lower_range, power=1,
78 attraction_enabled=True, emergency_strength=EMERGENCY_REPULSION_STRENGTH):
79 """@brief Full pairwise force scalar: emergency repulsion + standard gravity.
80
81 Emergency anti-stacking repulsion dominates when @p distance is below
82 @p lower_range * 2 (always on, even with attraction disabled). Otherwise,
83 when @p attraction_enabled, the standard gravity force applies with
84 @p coefficient; with attraction off and outside the emergency range, the
85 force is zero.
86
87 @param distance Center-to-center distance between the two bodies.
88 @param coefficient Attract/repel coefficient g (see interaction_coefficient()).
89 @param count Number of interacting bodies (normalizes the gravity force).
90 @param lower_range Emergency repulsion triggers below 2 * this distance.
91 @param power Exponent on distance for the gravity force.
92 @param attraction_enabled When False, only emergency repulsion can act.
93 @param emergency_strength Magnitude of the emergency repulsion.
94 @return float force scalar (multiply by the dx/dy delta to get the vector).
95 """
96 if distance < lower_range * 2:
97 return emergency_repulsion(emergency_strength)
98 if attraction_enabled:
99 return gravity_force(coefficient, distance, count, power)
100 return 0.0

103def soft_obstacle_push(distance, edge, soft_zone, soft_strength):
104 """@brief Quadratic pre-contact obstacle repulsion magnitude.
105
106 Zero outside the soft zone; rises quadratically from the zone's outer edge to
107 full @p soft_strength at the physical edge. Multiply by the outward unit
108 vector (dx/dist, dy/dist) to get the push vector.
109
110 @param distance Body-to-obstacle-center distance.
111 @param edge Obstacle radius + body radius (the physical contact distance).
112 @param soft_zone Width (px) of the pre-contact zone beyond @p edge.
113 @param soft_strength Push magnitude at the physical edge.
114 @return float push magnitude (0.0 outside the soft zone).
115 """
116 if soft_zone <= 0 or distance >= edge + soft_zone:
117 return 0.0
118 t = max(0.0, (edge + soft_zone - distance) / soft_zone)
119 return soft_strength * t * t

122def reflect_velocity(vx, vy, nx, ny, restitution=1.0):
123 """@brief Reflect a velocity off a surface with unit normal (nx, ny).
124
125 Only reflects when the body is moving into the surface (negative normal
126 component); otherwise the velocity is returned unchanged. @p restitution
127 scales the reflected speed (1.0 = elastic, < 1 = energy loss).
128
129 @param vx, vy Current velocity components.
130 @param nx, ny Unit surface normal pointing away from the surface.
131 @param restitution Bounce energy retention in [0, 1].
132 @return (vx, vy) the (possibly) reflected velocity.
133 """
134 dot = vx * nx + vy * ny
135 if dot < 0:
136 vx -= 2 * dot * nx
137 vy -= 2 * dot * ny
138 vx *= restitution
139 vy *= restitution
140 return vx, vy

143def clamp_speed(vx, vy, max_speed):
144 """@brief Cap a velocity vector's magnitude at @p max_speed (direction kept).
145
146 @param vx, vy Current velocity components.
147 @param max_speed Maximum allowed speed.
148 @return (vx, vy) the speed-limited velocity.
149 """
150 speed = (vx ** 2 + vy ** 2) ** 0.5
151 if speed > max_speed:
152 scale = max_speed / speed
153 vx *= scale
154 vy *= scale
155 return vx, vy
