1"""@file run_all.py
2@brief Run the 2x2 ablation (CFL on/off x emergent on/off) repeatedly.
3
4Each repetition launches all four configurations concurrently (one pygame window
5each), then waits for that batch of four to finish before starting the next
6repetition. Every run is written to its own folder:
7
8 `logs/<config>/<config>_run<NN>/`
9
10so the repeated runs of each configuration are grouped together for averaging.
11
12Usage (from the repo root):
13 python tools/run_all.py # RUNS repetitions (default below)
14 python tools/run_all.py 5 # override: 5 repetitions
15"""

25 "configs/cfl_off__emergent_off.json",
26]

28# Number of repetitions per configuration.
29RUNS = 10
30
31if __name__ == "__main__":

33 if len(sys.argv) > 1:

38

40
41 for i in range(1, runs + 1):

43 print(f"\n[run_all] === repetition {nn}/{runs}: launching "

49 log_dir = f"logs/{name}"

54 cwd=root,

61 # Wait for this batch of four to finish before the next repetition.
